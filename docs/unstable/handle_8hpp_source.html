86 template <typename T, Allocator A = mallocator<>>
87 struct handle_heap {
88 static_assert(std::is_nothrow_destructible_v<T>, "nda::mem::handle_heap requires the value_type to have a non-throwing destructor");
89
90 private:
91 // Pointer to the start of the actual data.
92 T *_data = nullptr;
93
94 // Size of the data (number of T elements). Invariant: size > 0 iif data != nullptr.
95 size_t _size = 0;
96
97 // Allocator to use.
98#ifndef NDA_DEBUG_LEAK_CHECK
99 static inline A allocator; // NOLINT (allocator is not specific to a single instance)
100#else
101 static inline leak_check<A> allocator; // NOLINT (allocator is not specific to a single instance)
102#endif
103
104 // For shared ownership (points to a blk_T_t).
105 mutable std::shared_ptr<void> sptr;
106
107 // Type of the memory block, i.e. a pointer to the data and its size.
108 using blk_T_t = std::pair<T *, size_t>;
109
110 // Release the handled memory (data pointer and size are not set to null here).
111 static void destruct(blk_T_t b) noexcept {
112 auto [data, size] = b;
113
114 // do nothing if the data is null
115 if (data == nullptr) return;
116
117 // if needed, call the destructors of the objects stored
118 if constexpr (A::address_space == Host and !(std::is_trivial_v<T> or nda::is_complex_v<T>)) {
119 for (size_t i = 0; i < size; ++i) data[i].~T();
120 }
121
122 // deallocate the memory block
123 allocator.deallocate({(char *)data, size * sizeof(T)});
124 }
125
126 // Deleter for the shared pointer.
127 static void deleter(void *p) noexcept { destruct(*((blk_T_t *)p)); }
128
129 public:
131 using value_type = T;
132
134 using allocator_type = A;
135
137 static constexpr auto address_space = allocator_type::address_space;
138
139 static constexpr bool is_aligned = allocator_type::is_aligned;
140
145 std::shared_ptr<void> get_sptr() const {
146 if (not sptr) sptr.reset(new blk_T_t{_data, _size}, deleter);
147 return sptr;
148 }
149
155 ~handle_heap() noexcept {
156 if (not sptr and not(is_null())) destruct({_data, _size});
157 }
158
160 handle_heap() = default;
161
166 handle_heap(handle_heap &&h) noexcept : _data(h._data), _size(h._size), sptr(std::move(h.sptr)) {
167 h._data = nullptr;
168 h._size = 0;
169 }
170
177 handle_heap &operator=(handle_heap &&h) noexcept {
178 // release current resources if they are not shared and not null
179 if (not sptr and not(is_null())) destruct({_data, _size});
180
181 // move the resources from the source handle
182 _data = h._data;
183 _size = h._size;
184 sptr = std::move(h.sptr);
185
186 // reset the source handle to a null state
187 h._data = nullptr;
188 h._size = 0;
189 return *this;
190 }
191
196 explicit handle_heap(handle_heap const &h) : handle_heap(h.size(), do_not_initialize) {
197 if (is_null()) return;
198 if constexpr (std::is_trivially_copyable_v<T>) {
199 memcpy<address_space, address_space>(_data, h.data(), h.size() * sizeof(T));
200 } else {
201 for (size_t i = 0; i < _size; ++i) new (_data + i) T(h[i]);
202 }
203 }

211 handle_heap &operator=(handle_heap const &h) {
212 *this = handle_heap{h};
213 return *this;

224 if (is_null()) return;
225 if constexpr (std::is_trivially_copyable_v<T>) {
226 memcpy<address_space, H::address_space>((void *)_data, (void *)h.data(), _size * sizeof(T));
227 } else {
228 static_assert(address_space == H::address_space,
229 "Constructing an nda::mem::handle_heap from a handle of a different address space requires a trivially copyable value_type");

252 if (size == 0) return;

254 if (not b.ptr) throw std::bad_alloc{};

257 }

263 handle_heap(long size, init_zero_t) {
264 if (size == 0) return;
265 auto b = allocator.allocate_zero(size * sizeof(T), type_alignment_info<T>::required_alignment);
266 if (not b.ptr) throw std::bad_alloc{};

282 handle_heap(long size) {
283 if (size == 0) return;
284 blk_t b;
285 if constexpr (is_complex_v<T> && init_dcmplx)
286 b = allocator.allocate_zero(size * sizeof(T), type_alignment_info<T>::required_alignment);
287 else
288 b = allocator.allocate(size * sizeof(T), type_alignment_info<T>::required_alignment);
289 if (not b.ptr) throw std::bad_alloc{};
290 _data = (T *)b.ptr;
291 _size = size;
292
293 // call placement new for non trivial and non complex types
294 if constexpr (!std::is_trivial_v<T> and !is_complex_v<T>) {
295 for (size_t i = 0; i < size; ++i) new (_data + i) T();
296 }
297 }

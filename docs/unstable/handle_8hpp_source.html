99 template <typename T, Allocator A = mallocator<>>
100 struct handle_heap {
101 static_assert(std::is_nothrow_destructible_v<T>, "nda::mem::handle_heap requires the value_type to have a non-throwing destructor");
102
103 private:
104 // Pointer to the start of the actual data.
105 T *_data = nullptr;
106
107 // Size of the data (number of T elements). Invariant: size > 0 iif data != nullptr.
108 size_t _size = 0;
109
110 // Allocator to use.
111#ifndef NDA_DEBUG_LEAK_CHECK
112 static inline A allocator; // NOLINT (allocator is not specific to a single instance)
113#else
114 static inline leak_check<A> allocator; // NOLINT (allocator is not specific to a single instance)
115#endif
116
117 // For shared ownership (points to a blk_T_t).
118 mutable std::shared_ptr<void> sptr;
119
120 // Type of the memory block, i.e. a pointer to the data and its size.
121 using blk_T_t = std::pair<T *, size_t>;
122
123 // Release the handled memory (data pointer and size are not set to null here).
124 static void destruct(blk_T_t b) noexcept {
125 auto [data, size] = b;
126
127 // do nothing if the data is null
128 if (data == nullptr) return;
129
130 // if needed, call the destructors of the objects stored
131 if constexpr (A::address_space == Host and !(std::is_trivial_v<T> or nda::is_complex_v<T>)) {
132 for (size_t i = 0; i < size; ++i) data[i].~T();
133 }
134
135 // deallocate the memory block
136 allocator.deallocate({(char *)data, size * sizeof(T)});
137 }
138
139 // Deleter for the shared pointer.
140 static void deleter(void *p) noexcept { destruct(*((blk_T_t *)p)); }
141
142 public:
144 using value_type = T;
145
147 using allocator_type = A;
148
150 static constexpr auto address_space = allocator_type::address_space;
151
152 static constexpr bool is_aligned = allocator_type::is_aligned;
153
158 std::shared_ptr<void> get_sptr() const {
159 if (not sptr) sptr.reset(new blk_T_t{_data, _size}, deleter);
160 return sptr;
161 }
162
168 ~handle_heap() noexcept {
169 if (not sptr and not(is_null())) destruct({_data, _size});
170 }
171
173 handle_heap() = default;
174
179 handle_heap(handle_heap &&h) noexcept : _data(h._data), _size(h._size), sptr(std::move(h.sptr)) {
180 h._data = nullptr;
181 h._size = 0;
182 }
183
190 handle_heap &operator=(handle_heap &&h) noexcept {
191 // release current resources if they are not shared and not null
192 if (not sptr and not(is_null())) destruct({_data, _size});
193
194 // move the resources from the source handle
195 _data = h._data;
196 _size = h._size;
197 sptr = std::move(h.sptr);
198
199 // reset the source handle to a null state
200 h._data = nullptr;
201 h._size = 0;
202 return *this;
203 }
204
209 explicit handle_heap(handle_heap const &h) : handle_heap(h.size(), do_not_initialize) {
210 if (is_null()) return;
211 if constexpr (std::is_trivially_copyable_v<T>) {
212 memcpy<address_space, address_space>(_data, h.data(), h.size() * sizeof(T));
213 } else {
214 for (size_t i = 0; i < _size; ++i) new (_data + i) T(h[i]);
215 }
216 }
217
224 handle_heap &operator=(handle_heap const &h) {
225 *this = handle_heap{h};
226 return *this;

295 handle_heap(long size) {
296 if (size == 0) return;
297 blk_t b;
298 if constexpr (is_complex_v<T> && init_dcmplx)
299 b = allocator.allocate_zero(size * sizeof(T), type_alignment_info<T>::required_alignment);
300 else

302 if (not b.ptr) throw std::bad_alloc{};
303 _data = (T *)b.ptr;
304 _size = size;
305
306 // call placement new for non trivial and non complex types
307 if constexpr (!std::is_trivial_v<T> and !is_complex_v<T>) {
308 for (size_t i = 0; i < size; ++i) new (_data + i) T();
309 }
310 }

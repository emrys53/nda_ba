99 template <typename T, Allocator A = mallocator<>>
100 struct handle_heap {
101 static_assert(std::is_nothrow_destructible_v<T>, "nda::mem::handle_heap requires the value_type to have a non-throwing destructor");
102
103 // TODO: change the concept of Allocator to accommodate width and add widths.
104 static constexpr size_t padding = A:: template padding<T>;
105
106 private:
107 // Pointer to the start of the actual data.
108 T *_data = nullptr;
109
110 // Size of the data (number of T elements). Invariant: size > 0 iif data != nullptr.
111 size_t _size = 0;
112
113 // Allocator to use.
114#ifndef NDA_DEBUG_LEAK_CHECK
115 static inline A allocator; // NOLINT (allocator is not specific to a single instance)
116#else
117 static inline leak_check<A> allocator; // NOLINT (allocator is not specific to a single instance)
118#endif
119
120 // For shared ownership (points to a blk_T_t).
121 mutable std::shared_ptr<void> sptr;
122
123 // Type of the memory block, i.e. a pointer to the data and its size.
124 using blk_T_t = std::pair<T *, size_t>;
125
126 // Release the handled memory (data pointer and size are not set to null here).
127 static void destruct(blk_T_t b) noexcept {
128 auto [data, size] = b;
129
130 // do nothing if the data is null
131 if (data == nullptr) return;
132
133 // if needed, call the destructors of the objects stored
134 if constexpr (A::address_space == Host and !(std::is_trivial_v<T> or nda::is_complex_v<T>)) {
135 for (size_t i = 0; i < size; ++i) data[i].~T();
136 }
137
138 // deallocate the memory block
139 allocator.deallocate({(char *)data, size * sizeof(T)});
140 }
141
142 // Deleter for the shared pointer.
143 static void deleter(void *p) noexcept { destruct(*((blk_T_t *)p)); }
144
145 public:
147 using value_type = T;
148
150 using allocator_type = A;
151
153 static constexpr auto address_space = allocator_type::address_space;
154
159 std::shared_ptr<void> get_sptr() const {
160 if (not sptr) sptr.reset(new blk_T_t{_data, _size}, deleter);
161 return sptr;
162 }
163
169 ~handle_heap() noexcept {
170 if (not sptr and not(is_null())) destruct({_data, _size});
171 }
172
174 handle_heap() = default;
175
180 handle_heap(handle_heap &&h) noexcept : _data(h._data), _size(h._size), sptr(std::move(h.sptr)) {
181 h._data = nullptr;
182 h._size = 0;
183 }
184
191 handle_heap &operator=(handle_heap &&h) noexcept {
192 // release current resources if they are not shared and not null
193 if (not sptr and not(is_null())) destruct({_data, _size});
194
195 // move the resources from the source handle
196 _data = h._data;
197 _size = h._size;
198 sptr = std::move(h.sptr);
199
200 // reset the source handle to a null state
201 h._data = nullptr;
202 h._size = 0;
203 return *this;
204 }

210 explicit handle_heap(handle_heap const &h) : handle_heap(h.size(), do_not_initialize) {
211 if (is_null()) return;
212 if constexpr (std::is_trivially_copyable_v<T>) {
213 memcpy<address_space, address_space>(_data, h.data(), h.size() * sizeof(T));
214 } else {
215 for (size_t i = 0; i < _size; ++i) new (_data + i) T(h[i]);
216 }

296 handle_heap(long size) {
297 if (size == 0) return;
298 blk_t b;
299 if constexpr (is_complex_v<T> && init_dcmplx)
300 b = allocator.allocate_zero(size * sizeof(T), type_alignment_info<T>::required_alignment);
301 else

303 if (not b.ptr) throw std::bad_alloc{};
304 _data = (T *)b.ptr;
305 _size = size;
306
307 // call placement new for non trivial and non complex types
308 if constexpr (!std::is_trivial_v<T> and !is_complex_v<T>) {
309 for (size_t i = 0; i < size; ++i) new (_data + i) T();
310 }
311 }

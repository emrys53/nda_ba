60 template <Array A, typename F, typename R>
61 auto fold(F f, A const &a, R r) {
62 // cast the initial value to the return type of f to avoid narrowing
63 using res_t = std::decay_t<decltype(make_regular(f(r, get_value_t<A>{})))>;
64 auto res = res_t{r};
65 nda::for_each(a.shape(), [&a, &res, &f](auto &&...args) { res = f(res, a(args...)); });
66 return res;
67 }

75 template <Array A, typename F_SIMD, typename F_SCALAR, Vectorizable R>
76 requires(vectorizable_array<A> and std::is_same_v<R, get_value_t<A>>)
77 auto fold(F_SIMD f_simd, F_SCALAR f_scalar, A const &a, native_simd<R> r_simd, R r_scalar) {
78 nda::for_each_static<0, get_layout_info<A>.stride_order>(
79 a.shape(), [&a, &r_simd, &f_simd](auto &&...args) { r_simd = f_simd(r_simd, native_simd<R>(a.load(args...))); },
80 [&a, &r_scalar, &f_scalar](auto &&...args) { r_scalar = f_scalar(r_scalar, a(args...)); }, native_simd<R>::size());
81 alignas(r_simd.alignment()) std::array<R, r_simd.size()> res;
82 r_simd.store(res.data());
83 for (int i = 0; i < r_simd.size(); i++) { r_scalar = f_scalar(r_scalar, res[i]); }
84 return r_scalar;
85 }

143 template <Array A>
144 auto max_element(A const &a) {
145 if constexpr (vectorizable_array<A>) {
146 using value_t = get_value_t<A>;
147 using simd_t = native_simd<value_t>;
148 simd_t max_simd(get_first_element(a));
149 auto f_simd = [&a, &max_simd](auto &&...args) { max_simd = simd::max(max_simd, a.load(args...)); };
150 value_t max_scalar = get_first_element(a);
151 auto f_scalar = [&a, &max_scalar](auto &&...args) { max_scalar = std::max(max_scalar, a(args...)); };
152 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
153 return std::max(max_scalar, simd::reduce_max(max_simd));
154 }
155 return fold(
156 [](auto const &x, auto const &y) {
157 using std::max;
158 return max(x, y);
159 },
160 a, get_first_element(a));
161 }

201 template <ArrayOfRank<2> A>
202 double frobenius_norm(A const &a) {
203 if constexpr (vectorizable_array<A> and not is_complex_v<get_value_t<A>>) {
204 using value_t = get_value_t<A>;
205 using simd_t = native_simd<value_t>;
206 simd_t r_simd(value_t(0));
207 auto f_simd = [&a, &r_simd](auto &&...args) {
208 simd_t x = a.load(args...);
209 simd_t abs = simd::abs(x);
210 if constexpr (std::is_integral_v<value_t>) {
211 r_simd = abs * abs + r_simd;
212 } else {
213 r_simd = simd::fma_add(abs, abs, r_simd);
214 }
215 };
216 double r = 0;
217 auto f_scalar = [&a, &r](auto &&...args) {
218 auto abs = std::abs(a(args...));
219 r = abs * abs + r;
220 };
221 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
222 return std::sqrt((static_cast<double>(simd::reduce_sum(r_simd)) + r));
223 }
224 return std::sqrt(fold(
225 [](double r, auto const &x) -> double {
226 auto ab = std::abs(x);
227 return r + ab * ab;
228 },
229 a, double(0)));
230 }

239 template <Array A, typename Value = get_value_t<A>>
240 auto sum(A const &a)
241 requires(nda::Scalar<Value> or nda::Array<Value>)
242 {
243 if constexpr (nda::Scalar<Value>) {
244 if constexpr (vectorizable_array<A>) {
245 using value_t = get_value_t<A>;
246 using simd_t = native_simd<value_t>;
247 simd_t sum_simd(value_t{0});
248 auto f_simd = [&a, &sum_simd](auto &&...args) { sum_simd += a.load(args...); };
249 value_t sum_scalar{0};
250 auto f_scalar = [&a, &sum_scalar](auto &&...args) { sum_scalar += a(args...); };
251 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
252 return sum_scalar + simd::reduce_sum(sum_simd);
253 }
254 return fold(std::plus<>{}, a);
255 } else { // Array<Value>
256 return fold(std::plus<>{}, a, Value::zeros(get_first_element(a).shape()));
257 }
258 }

267 template <Array A, typename Value = get_value_t<A>>
268 auto product(A const &a)
269 requires(nda::Scalar<Value> or nda::Array<Value>)
270 {
271 if constexpr (nda::Scalar<Value>) {
272 if constexpr (vectorizable_array<A>) {
273 using value_t = get_value_t<A>;
274 using simd_t = native_simd<value_t>;
275 simd_t product_simd(value_t{1});
276 auto f_simd = [&a, &product_simd](auto &&...args) { product_simd *= a.load(args...); };
277 value_t product_scalar{1};
278 auto f_scalar = [&a, &product_scalar](auto &&...args) { product_scalar *= a(args...); };
279 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
280 return product_scalar * simd::reduce_mul(product_simd);
281 }
282 return fold(std::multiplies<>{}, a, get_value_t<A>{1});
283 } else { // Array<Value>
284 return fold(std::multiplies<>{}, a, Value::ones(get_first_element(a).shape()));
285 }

297 template <Array A, Array B>
298 requires(nda::get_rank<A> == nda::get_rank<B>)
299 [[nodiscard]] constexpr auto hadamard(A &&a, B &&b) {
300 if constexpr (vectorizable_array<A> and vectorizable_array<B> and std::is_same_v<get_value_t<A>, get_value_t<B>>) {
301 using value_t = get_value_t<A>;
302 using simd_t = native_simd<value_t>;
303 struct mul {
304 value_t operator()(const value_t &x, const value_t &y) const { return x * y; }
305 simd_t load(const simd_t &x, const simd_t &y) const { return x * y; };
306 };
307 return nda::map(mul{})(std::forward<A>(a), std::forward<B>(b));
308 } else {
309 return nda::map([](auto const &x, auto const &y) { return x * y; })(std::forward<A>(a), std::forward<B>(b));
310 }
311 }

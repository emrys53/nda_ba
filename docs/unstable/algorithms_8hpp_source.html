71 template <Array A, typename F, typename R>
72 auto fold(F f, A const &a, R r) {
73 // cast the initial value to the return type of f to avoid narrowing
74 using res_t = std::decay_t<decltype(make_regular(f(r, get_value_t<A>{})))>;
75 auto res = res_t{r};
76 nda::for_each(a.shape(), [&a, &res, &f](auto &&...args) { res = f(res, a(args...)); });
77 return res;
78 }

86 template <bool isAligned, Array A, typename F_SCALAR, typename F_SIMD, Vectorizable R>
87 auto fold(F_SIMD f_simd, F_SCALAR f_scalar, A const &a, native_simd<R> r_simd, R r_scalar) {
88 nda::for_each(
89 a.shape(),
90 [&a, &r_simd, &f_simd](auto &&...args) {
91 if constexpr (isAligned) {
92 r_simd = f_simd(r_simd, native_simd<R>(&a(args...)));
93 } else {
94 native_simd<R> tmp;
95 tmp.load_unaligned(&a(args...));
96 r_simd = f_simd(r_simd, tmp);
97 }
98 },
99 [&a, &r_scalar, &f_scalar](auto &&...args) { r_scalar = f_scalar(r_scalar, a(args...)); }, native_simd<R>::size());
100 alignas(r_simd.alignment()) std::array<R, r_simd.size()> res;
101 r_simd.store(res.data());
102 for (int i = 0; i < r_simd.size(); i++) { r_scalar = f_scalar(r_scalar, res[i]); }
103 return r_scalar;
104 }

162 template <Array A>
163 auto max_element(A const &a) {
164 if constexpr (vectorizable_array<A>) {
165 using value_t = get_value_t<A>;
166 using simd_t = native_simd<value_t>;
167 simd_t max_simd(get_first_element(a));
168 auto f_simd = [&a, &max_simd](auto &&...args) { max_simd = simd::max(max_simd, a.load(args...)); };
169 value_t max_scalar = get_first_element(a);
170 auto f_scalar = [&a, &max_scalar](auto &&...args) { max_scalar = std::max(max_scalar, a(args...)); };
171 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
172 return std::max(max_scalar, simd::reduce_max(max_simd));
173 }
174 return fold(
175 [](auto const &x, auto const &y) {
176 using std::max;
177 return max(x, y);
178 },
179 a, get_first_element(a));
180 }

220 template <ArrayOfRank<2> A>
221 double frobenius_norm(A const &a) {
222 return std::sqrt(fold(
223 [](double r, auto const &x) -> double {
224 auto ab = std::abs(x);
225 return r + ab * ab;
226 },
227 a, double(0)));

237 template <Array A, typename Value = get_value_t<A>>
238 auto sum(A const &a)
239 requires(nda::Scalar<Value> or nda::Array<Value>)
240 {
241 if constexpr (nda::Scalar<Value>) {
242 if constexpr (vectorizable_array<A>) {
243 using value_t = get_value_t<A>;
244 using simd_t = native_simd<value_t>;
245 simd_t sum_simd(value_t{0});
246 auto f_simd = [&a, &sum_simd](auto &&...args) { sum_simd += a.load(args...); };
247 value_t sum_scalar{0};
248 auto f_scalar = [&a, &sum_scalar](auto &&...args) { sum_scalar += a(args...); };
249 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
250 return sum_scalar + simd::reduce_sum(sum_simd);
251 }
252 return fold(std::plus<>{}, a);
253 } else { // Array<Value>
254 return fold(std::plus<>{}, a, Value::zeros(get_first_element(a).shape()));
255 }
256 }

265 template <Array A, typename Value = get_value_t<A>>
266 auto product(A const &a)
267 requires(nda::Scalar<Value> or nda::Array<Value>)
268 {
269 if constexpr (nda::Scalar<Value>) {
270 if constexpr (vectorizable_array<A>) {
271 using value_t = get_value_t<A>;
272 using simd_t = native_simd<value_t>;
273 simd_t product_simd(value_t{1});
274 auto f_simd = [&a, &product_simd](auto &&...args) { product_simd *= a.load(args...); };
275 value_t product_scalar{1};
276 auto f_scalar = [&a, &product_scalar](auto &&...args) { product_scalar *= a(args...); };
277 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
278 return product_scalar * simd::reduce_mul(product_simd);
279 }
280 return fold(std::multiplies<>{}, a, get_value_t<A>{1});
281 } else { // Array<Value>
282 return fold(std::multiplies<>{}, a, Value::ones(get_first_element(a).shape()));
283 }

295 template <Array A, Array B>
296 requires(nda::get_rank<A> == nda::get_rank<B>)
297 [[nodiscard]] constexpr auto hadamard(A &&a, B &&b) {
298 if constexpr (vectorizable_array<A> and vectorizable_array<B> and std::is_same_v<get_value_t<A>, get_value_t<B>>) {
299 using value_t = get_value_t<A>;
300 using simd_t = native_simd<value_t>;
301 struct mul {
302 value_t operator()(const value_t &x, const value_t &y) const { return x * y; }
303 simd_t load(const simd_t &x, const simd_t &y) const { return x * y; };
304 };
305 return nda::map(mul{})(std::forward<A>(a), std::forward<B>(b));
306 } else {
307 return nda::map([](auto const &x, auto const &y) { return x * y; })(std::forward<A>(a), std::forward<B>(b));
308 }
309 }

338 EXPECTS(a.size() == b.size());

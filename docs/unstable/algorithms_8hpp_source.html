71 template <Array A, typename F, typename R>
72 auto fold(F f, A const &a, R r) {
73 // cast the initial value to the return type of f to avoid narrowing
74 using res_t = std::decay_t<decltype(make_regular(f(r, get_value_t<A>{})))>;
75 auto res = res_t{r};
76 nda::for_each(a.shape(), [&a, &res, &f](auto &&...args) { res = f(res, a(args...)); });
77 return res;
78 }

86 template <Array A, typename F_SIMD, typename F_SCALAR, Vectorizable R>
87 requires(vectorizable_array<A> and std::is_same_v<R, get_value_t<A>>)
88 auto fold(F_SIMD f_simd, F_SCALAR f_scalar, A const &a, native_simd<R> r_simd, R r_scalar) {
89 nda::for_each_static<0, get_layout_info<A>.stride_order>(
90 a.shape(), [&a, &r_simd, &f_simd](auto &&...args) { r_simd = f_simd(r_simd, native_simd<R>(a.load(args...))); },
91 [&a, &r_scalar, &f_scalar](auto &&...args) { r_scalar = f_scalar(r_scalar, a(args...)); }, native_simd<R>::size());
92 alignas(r_simd.alignment()) std::array<R, r_simd.size()> res;
93 r_simd.store(res.data());
94 for (int i = 0; i < r_simd.size(); i++) { r_scalar = f_scalar(r_scalar, res[i]); }
95 return r_scalar;
96 }

154 template <Array A>
155 auto max_element(A const &a) {
156 if constexpr (vectorizable_array<A>) {
157 using value_t = get_value_t<A>;
158 using simd_t = native_simd<value_t>;
159 simd_t max_simd(get_first_element(a));
160 auto f_simd = [&a, &max_simd](auto &&...args) { max_simd = simd::max(max_simd, a.load(args...)); };
161 value_t max_scalar = get_first_element(a);
162 auto f_scalar = [&a, &max_scalar](auto &&...args) { max_scalar = std::max(max_scalar, a(args...)); };
163 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
164 return std::max(max_scalar, simd::reduce_max(max_simd));
165 }
166 return fold(
167 [](auto const &x, auto const &y) {
168 using std::max;
169 return max(x, y);
170 },
171 a, get_first_element(a));
172 }

212 template <ArrayOfRank<2> A>
213 double frobenius_norm(A const &a) {
214 if constexpr (vectorizable_array<A> and not is_complex_v<get_value_t<A>>) {
215 using value_t = get_value_t<A>;
216 using simd_t = native_simd<value_t>;
217 simd_t r_simd(value_t(0));
218 auto f_simd = [&a, &r_simd](auto &&...args) {
219 simd_t x = a.load(args...);
220 simd_t abs = simd::abs(x);
221 if constexpr (std::is_integral_v<value_t>) {
222 r_simd = abs * abs + r_simd;
223 } else {
224 r_simd = simd::fma_add(abs, abs, r_simd);
225 }
226 };
227 double r = 0;
228 auto f_scalar = [&a, &r](auto &&...args) {
229 auto abs = std::abs(a(args...));
230 r = abs * abs + r;
231 };
232 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
233 return std::sqrt((static_cast<double>(simd::reduce_sum(r_simd)) + r));
234 }
235 return std::sqrt(fold(
236 [](double r, auto const &x) -> double {
237 auto ab = std::abs(x);
238 return r + ab * ab;
239 },
240 a, double(0)));
241 }

250 template <Array A, typename Value = get_value_t<A>>
251 auto sum(A const &a)
252 requires(nda::Scalar<Value> or nda::Array<Value>)
253 {
254 if constexpr (nda::Scalar<Value>) {
255 if constexpr (vectorizable_array<A>) {
256 using value_t = get_value_t<A>;
257 using simd_t = native_simd<value_t>;
258 simd_t sum_simd(value_t{0});
259 auto f_simd = [&a, &sum_simd](auto &&...args) { sum_simd += a.load(args...); };
260 value_t sum_scalar{0};
261 auto f_scalar = [&a, &sum_scalar](auto &&...args) { sum_scalar += a(args...); };
262 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
263 return sum_scalar + simd::reduce_sum(sum_simd);
264 }
265 return fold(std::plus<>{}, a);
266 } else { // Array<Value>
267 return fold(std::plus<>{}, a, Value::zeros(get_first_element(a).shape()));
268 }
269 }

278 template <Array A, typename Value = get_value_t<A>>
279 auto product(A const &a)
280 requires(nda::Scalar<Value> or nda::Array<Value>)
281 {
282 if constexpr (nda::Scalar<Value>) {
283 if constexpr (vectorizable_array<A>) {
284 using value_t = get_value_t<A>;
285 using simd_t = native_simd<value_t>;
286 simd_t product_simd(value_t{1});
287 auto f_simd = [&a, &product_simd](auto &&...args) { product_simd *= a.load(args...); };
288 value_t product_scalar{1};
289 auto f_scalar = [&a, &product_scalar](auto &&...args) { product_scalar *= a(args...); };
290 nda::for_each_static<0, get_layout_info<A>.stride_order>(a.shape(), std::move(f_simd), std::move(f_scalar), simd_t::size());
291 return product_scalar * simd::reduce_mul(product_simd);
292 }
293 return fold(std::multiplies<>{}, a, get_value_t<A>{1});
294 } else { // Array<Value>
295 return fold(std::multiplies<>{}, a, Value::ones(get_first_element(a).shape()));
296 }

308 template <Array A, Array B>
309 requires(nda::get_rank<A> == nda::get_rank<B>)
310 [[nodiscard]] constexpr auto hadamard(A &&a, B &&b) {
311 if constexpr (vectorizable_array<A> and vectorizable_array<B> and std::is_same_v<get_value_t<A>, get_value_t<B>>) {
312 using value_t = get_value_t<A>;
313 using simd_t = native_simd<value_t>;
314 struct mul {
315 value_t operator()(const value_t &x, const value_t &y) const { return x * y; }
316 simd_t load(const simd_t &x, const simd_t &y) const { return x * y; };
317 };
318 return nda::map(mul{})(std::forward<A>(a), std::forward<B>(b));
319 } else {
320 return nda::map([](auto const &x, auto const &y) { return x * y; })(std::forward<A>(a), std::forward<B>(b));
321 }
322 }

350 using TU = decltype(std::declval<T>() * std::declval<U>());
351 EXPECTS(a.size() == b.size());

71 template <Array A, typename F, typename R>
72 auto fold(F f, A const &a, R r) {
73 // cast the initial value to the return type of f to avoid narrowing
74 using res_t = std::decay_t<decltype(make_regular(f(r, get_value_t<A>{})))>;
75 auto res = res_t{r};
76 nda::for_each(a.shape(), [&a, &res, &f](auto &&...args) { res = f(res, a(args...)); });
77 return res;
78 }

86 template <bool isAligned, Array A, typename F_SCALAR, typename F_SIMD, Vectorizable R>
87 auto fold(F_SIMD f_simd, F_SCALAR f_scalar, A const &a, native_simd<R> r_simd, R r_scalar) {
88 nda::for_each(
89 a.shape(),
90 [&a, &r_simd, &f_simd](auto &&...args) {
91 native_simd<R> tmp;
92 if constexpr (isAligned) {
93 tmp.load(&a(args...));
94 } else {
95 tmp.load_unaligned(&a(args...));
96 }
97
98 r_simd = f_simd(r_simd, tmp);
99 },
100 [&a, &r_scalar, &f_scalar](auto &&...args) { r_scalar = f_scalar(r_scalar, a(args...)); }, native_simd<R>::size());
101 alignas(r_simd.alignment()) std::array<R, r_simd.size()> res;
102 r_simd.store(res.data());
103 for (int i = 0; i < r_simd.size(); i++) { r_scalar = f_scalar(r_scalar, res[i]); }
104 return r_scalar;
105 }

200 template <ArrayOfRank<2> A>
201 double frobenius_norm(A const &a) {
202 return std::sqrt(fold(
203 [](double r, auto const &x) -> double {
204 auto ab = std::abs(x);
205 return r + ab * ab;
206 },
207 a, double(0)));

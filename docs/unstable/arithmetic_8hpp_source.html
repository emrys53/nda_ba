186 template <typename... Args>
187 auto operator()(Args const &...args) const {
188 // addition
189 if constexpr (OP == '+') {
190 if constexpr (l_is_scalar) {
191 // lhs is a scalar
192 if constexpr (algebra == 'M')
193 // rhs is a matrix
194 return (std::equal_to{}(args...) ? l + r(args...) : r(args...));
195 else
196 // rhs is an array
197 return l + r(args...);
198 } else if constexpr (r_is_scalar) {
199 // rhs is a scalar
200 if constexpr (algebra == 'M')
201 // lhs is a matrix
202 return (std::equal_to{}(args...) ? l(args...) + r : l(args...));
203 else
204 // lhs is an array
205 return l(args...) + r;
206 } else
207 // both are arrays or matrices
208 return l(args...) + r(args...);
209 }
210
211 // subtraction
212 if constexpr (OP == '-') {
213 if constexpr (l_is_scalar) {
214 // lhs is a scalar
215 if constexpr (algebra == 'M')
216 // rhs is a matrix
217 return (std::equal_to{}(args...) ? l - r(args...) : -r(args...));
218 else
219 // rhs is an array
220 return l - r(args...);
221 } else if constexpr (r_is_scalar) {
222 // rhs is a scalar
223 if constexpr (algebra == 'M')
224 // lhs is a matrix
225 return (std::equal_to{}(args...) ? l(args...) - r : l(args...));
226 else
227 // lhs is an array
228 return l(args...) - r;
229 } else
230 // both are arrays or matrices
231 return l(args...) - r(args...);
232 }
233
234 // multiplication
235 if constexpr (OP == '*') {
236 if constexpr (l_is_scalar)
237 // lhs is a scalar
238 return l * r(args...);
239 else if constexpr (r_is_scalar)
240 // rhs is a scalar
241 return l(args...) * r;
242 else {
243 // both are arrays (matrix product is not supported here)
244 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
245 return l(args...) * r(args...);
246 }
247 }
248
249 // division
250 if constexpr (OP == '/') {
251 if constexpr (l_is_scalar) {
252 // lhs is a scalar
253 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
254 return l / r(args...);
255 } else if constexpr (r_is_scalar)
256 // rhs is a scalar
257 return l(args...) / r;
258 else {
259 // both are arrays (matrix division is not supported here)
260 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
261 return l(args...) / r(args...);
262 }
263 }

266 template <typename... Args>
267 auto load(Args const &...args) const {
268 auto diagonal_simd = [this](long i, long j) {
269 // This lambda function can only be used when we have a matrix. This constexpr is needed because otherwise we might get compile errors.
270 if constexpr (sizeof...(Args) == 2 and (Vectorizable<L_t> or Vectorizable<R_t>)) {
271 long diff = i - j;
272 if constexpr ((l_is_scalar and get_layout_info<R>.stride_order == Fortran_stride_order<2>)
273 or (r_is_scalar and get_layout_info<L>.stride_order == Fortran_stride_order<2>)) {
274 diff = -diff;
275 }
276 if constexpr (l_is_scalar) {
277 using simd_t = native_simd<L_t>;
278 if (diff < 0 or diff > simd_t::size() - 1) return r.load(i, j);
279 alignas(simd_t::alignment()) std::array<L_t, simd_t::size()> tmp{};
280 tmp[diff] = l;
281 if constexpr (OP == '+') {
282 return r.load(i, j) + simd_t(tmp.data());
283 } else {
284 return r.load(i, j) - simd_t(tmp.data());
285 }
286 } else if constexpr (r_is_scalar) {
287 using simd_t = native_simd<R_t>;
288 if (diff < 0 or diff > simd_t::size() - 1) return l.load(i, j);
289 alignas(simd_t::alignment()) std::array<R_t, simd_t::size()> tmp{};
290 tmp[diff] = r;
291 if constexpr (OP == '+') {
292 return l.load(i, j) + simd_t(tmp.data());
293 } else {
294 return l.load(i, j) - simd_t(tmp.data());
295 }
296 }
297 }
298 };
299 // addition
300 if constexpr (OP == '+') {
301 if constexpr (l_is_scalar) {
302 // lhs is a scalar
303 if constexpr (algebra == 'M')
304 // rhs is a matrix
305 return diagonal_simd(args...);
306 else
307 // rhs is an array
308 return native_simd<L_t>(l) + r.load(args...);
309 } else if constexpr (r_is_scalar) {
310 // rhs is a scalar
311 if constexpr (algebra == 'M') {
312 // lhs is a matrix
313 return diagonal_simd(args...);
314 } else
315 // lhs is an array
316 return l.load(args...) + native_simd<R_t>(r);
317 } else
318 // both are arrays or matrices
319 return l.load(args...) + r.load(args...);
320 }
321
322 // subtraction
323 if constexpr (OP == '-') {
324 if constexpr (l_is_scalar) {
325 // lhs is a scalar
326 if constexpr (algebra == 'M')
327 // rhs is a matrix
328 return diagonal_simd(args...);
329 else
330 // rhs is an array
331 return native_simd<L_t>(l) - r.load(args...);
332 } else if constexpr (r_is_scalar) {
333 // rhs is a scalar
334 if constexpr (algebra == 'M')
335 // lhs is a matrix
336 return diagonal_simd(args...);
337 else
338 // lhs is an array
339 return l.load(args...) - native_simd<R_t>(r);
340 } else
341 // both are arrays or matrices
342 return l.load(args...) - r.load(args...);
343 }
344
345 // multiplication
346 if constexpr (OP == '*') {
347 if constexpr (l_is_scalar)
348 // lhs is a scalar
349 return native_simd<L_t>(l) * r.load(args...);
350 else if constexpr (r_is_scalar)
351 // rhs is a scalar
352 return l.load(args...) * native_simd<R_t>(r);
353 else {
354 // both are arrays (matrix product is not supported here)

357 }
358 }

530 template <Array L, Array R>
531 auto operator*(L &&l, R &&r) {
532 // allowed algebras: A * A or M * M or M * V
533 static constexpr char l_algebra = get_algebra<L>;
534 static constexpr char r_algebra = get_algebra<R>;
535 static_assert(l_algebra != 'V', "Error in nda::operator*: Can not multiply vector by an array or a matrix");
536
537 // two arrays: A * A
538 if constexpr (l_algebra == 'A') {
539 static_assert(r_algebra == 'A', "Error in nda::operator*: Both types need to be arrays");
540 static_assert(get_rank<L> == get_rank<R>, "Error in nda::operator*: Rank mismatch");
541#ifdef NDA_ENFORCE_BOUNDCHECK
542 if (l.shape() != r.shape()) NDA_RUNTIME_ERROR << "Error in nda::operator*: Dimension mismatch: " << l.shape() << " != " << r.shape();
543#endif
544 return expr<'*', L, R>{std::forward<L>(l), std::forward<R>(r)};
545 }
546
547 // two matrices: M * M
548 if constexpr (l_algebra == 'M') {
549 static_assert(r_algebra != 'A', "Error in nda::operator*: Can not multiply a matrix by an array");
550 if constexpr (r_algebra == 'M')
551 // matrix * matrix
552 return matmul(std::forward<L>(l), std::forward<R>(r));
553 else
554 // matrix * vector
555 return matvecmul(std::forward<L>(l), std::forward<R>(r));
556 }
557 }

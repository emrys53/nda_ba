175 template <typename... Args>
176 auto operator()(Args const &...args) const {
177 // addition
178 if constexpr (OP == '+') {
179 if constexpr (l_is_scalar) {
180 // lhs is a scalar
181 if constexpr (algebra == 'M')
182 // rhs is a matrix
183 return (std::equal_to{}(args...) ? l + r(args...) : r(args...));
184 else
185 // rhs is an array
186 return l + r(args...);
187 } else if constexpr (r_is_scalar) {
188 // rhs is a scalar
189 if constexpr (algebra == 'M')
190 // lhs is a matrix
191 return (std::equal_to{}(args...) ? l(args...) + r : l(args...));
192 else
193 // lhs is an array
194 return l(args...) + r;
195 } else
196 // both are arrays or matrices
197 return l(args...) + r(args...);
198 }
199
200 // subtraction
201 if constexpr (OP == '-') {
202 if constexpr (l_is_scalar) {
203 // lhs is a scalar
204 if constexpr (algebra == 'M')
205 // rhs is a matrix
206 return (std::equal_to{}(args...) ? l - r(args...) : -r(args...));
207 else
208 // rhs is an array
209 return l - r(args...);
210 } else if constexpr (r_is_scalar) {
211 // rhs is a scalar
212 if constexpr (algebra == 'M')
213 // lhs is a matrix
214 return (std::equal_to{}(args...) ? l(args...) - r : l(args...));
215 else
216 // lhs is an array
217 return l(args...) - r;
218 } else
219 // both are arrays or matrices
220 return l(args...) - r(args...);
221 }
222
223 // multiplication
224 if constexpr (OP == '*') {
225 if constexpr (l_is_scalar)
226 // lhs is a scalar
227 return l * r(args...);
228 else if constexpr (r_is_scalar)
229 // rhs is a scalar
230 return l(args...) * r;
231 else {
232 // both are arrays (matrix product is not supported here)
233 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
234 return l(args...) * r(args...);
235 }
236 }
237
238 // division
239 if constexpr (OP == '/') {
240 if constexpr (l_is_scalar) {
241 // lhs is a scalar
242 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
243 return l / r(args...);
244 } else if constexpr (r_is_scalar)
245 // rhs is a scalar
246 return l(args...) / r;
247 else {
248 // both are arrays (matrix division is not supported here)
249 static_assert(algebra != 'M', "Error in nda::expr: Matrix algebra not supported");
250 return l(args...) / r(args...);
251 }
252 }

255 template <typename... Args>
256 auto load(Args const &...args) const {
257 auto diagonal_simd = [this](long i, long j) {
258 // This lambda function can only be used when we have a matrix. This constexpr is needed because otherwise we might get compile errors.
259 if constexpr (sizeof...(Args) == 2 and (Vectorizable<L_t> or Vectorizable<R_t>)) {
260 long diff = i - j;
261 if constexpr ((l_is_scalar and get_layout_info<R>.stride_order == Fortran_stride_order<2>)
262 or (r_is_scalar and get_layout_info<L>.stride_order == Fortran_stride_order<2>)) {
263 diff = -diff;
264 }
265 if constexpr (l_is_scalar) {
266 using simd_t = native_simd<L_t>;
267 if (diff < 0 or diff > simd_t::size() - 1) return r.load(i, j);
268 alignas(simd_t::alignment()) std::array<L_t, simd_t::size()> tmp{};
269 tmp[diff] = l;
270 if constexpr (OP == '+') {
271 return r.load(i, j) + simd_t(tmp.data(), simd_aligned_memory_t);
272 } else {
273 return r.load(i, j) - simd_t(tmp.data(), simd_aligned_memory_t);
274 }
275 } else if constexpr (r_is_scalar) {
276 using simd_t = native_simd<R_t>;
277 if (diff < 0 or diff > simd_t::size() - 1) return l.load(i, j);
278 alignas(simd_t::alignment()) std::array<R_t, simd_t::size()> tmp{};
279 tmp[diff] = r;
280 if constexpr (OP == '+') {
281 return l.load(i, j) + simd_t(tmp.data(), simd_aligned_memory_t);
282 } else {
283 return l.load(i, j) - simd_t(tmp.data(), simd_aligned_memory_t);
284 }
285 }
286 }
287 };

519 template <Array L, Array R>
520 auto operator*(L &&l, R &&r) {
521 // allowed algebras: A * A or M * M or M * V
522 static constexpr char l_algebra = get_algebra<L>;
523 static constexpr char r_algebra = get_algebra<R>;
524 static_assert(l_algebra != 'V', "Error in nda::operator*: Can not multiply vector by an array or a matrix");
525
526 // two arrays: A * A
527 if constexpr (l_algebra == 'A') {
528 static_assert(r_algebra == 'A', "Error in nda::operator*: Both types need to be arrays");
529 static_assert(get_rank<L> == get_rank<R>, "Error in nda::operator*: Rank mismatch");
530#ifdef NDA_ENFORCE_BOUNDCHECK
531 if (l.shape() != r.shape()) NDA_RUNTIME_ERROR << "Error in nda::operator*: Dimension mismatch: " << l.shape() << " != " << r.shape();
532#endif
533 return expr<'*', L, R>{std::forward<L>(l), std::forward<R>(r)};
534 }
535
536 // two matrices: M * M
537 if constexpr (l_algebra == 'M') {
538 static_assert(r_algebra != 'A', "Error in nda::operator*: Can not multiply a matrix by an array");
539 if constexpr (r_algebra == 'M')
540 // matrix * matrix
541 return matmul(std::forward<L>(l), std::forward<R>(r));
542 else
543 // matrix * vector
544 return matvecmul(std::forward<L>(l), std::forward<R>(r));
545 }
546 }

596 template <Array L, Array R>
597 Array auto operator/(L &&l, R &&r) {
598 // allowed algebras: A / A or M / M
599 static constexpr char l_algebra = get_algebra<L>;
600 static constexpr char r_algebra = get_algebra<R>;
601 static_assert(l_algebra != 'V', "Error in nda::operator/: Can not divide a vector by an array or a matrix");
602
603 // two arrays: A / A
604 if constexpr (l_algebra == 'A') {
605 static_assert(r_algebra == 'A', "Error in nda::operator/: Both types need to be arrays");

609#endif
610 return expr<'/', L, R>{std::forward<L>(l), std::forward<R>(r)};
611 }
612
613 // two matrices: M / M
614 if constexpr (l_algebra == 'M') {
615 static_assert(r_algebra == 'M', "Error in nda::operator*: Can not divide a matrix by an array/vector");
616 return std::forward<L>(l) * inverse(matrix<get_value_t<R>>{std::forward<R>(r)});

603template <char ResultAlgebra, bool SelfIsRvalue, typename Self, typename... Ts>
604FORCEINLINE static decltype(auto) call(Self &&self, Ts const &...idxs) noexcept(has_no_boundcheck) {
605 // resulting value type
606 using r_v_t = std::conditional_t<std::is_const_v<std::remove_reference_t<Self>>, ValueType const, ValueType>;
607
608 // behavior depends on the given arguments
609 if constexpr (clef::is_any_lazy<Ts...>) {
610 // if there are lazy arguments, e.g. as in A(i_) << i_, a lazy expression is returned
611 return clef::make_expr_call(std::forward<Self>(self), idxs...);
612 } else if constexpr (sizeof...(Ts) == 0) {
613 // if no arguments are given, a full view is returned
614 return basic_array_view<r_v_t, Rank, LayoutPolicy, Algebra, AccessorPolicy, OwningPolicy>{self.lay, self.sto};
615 } else {
616 // otherwise we check the arguments and either access a single element or make a slice
617 static_assert(((layout_t::template argument_is_allowed_for_call_or_slice<Ts> + ...) > 0),
618 "Error in array/view: Slice arguments must be convertible to range, ellipsis, or long (or string if the layout permits it)");
619
620 // number of arguments convertible to long
621 static constexpr int n_args_long = (layout_t::template argument_is_allowed_for_call<Ts> + ...);
622
623 if constexpr (n_args_long == rank) {
624 // access a single element
625 long offset = self.lay(idxs...);
626 if constexpr (is_view or not SelfIsRvalue) {
627 // if the calling object is a view or an lvalue, we return a reference
628 return AccessorPolicy::template accessor<r_v_t>::access(self.sto.data(), offset);
629 } else {
630 // otherwise, we return a copy of the value
631 return ValueType{self.sto[offset]};
632 }
633 } else {
634 // access a slice of the view/array
635 auto const [offset, idxm] = self.lay.slice(idxs...);
636 static constexpr auto res_rank = decltype(idxm)::rank();
637 // resulting algebra
638 static constexpr char newAlgebra = (ResultAlgebra == 'M' and (res_rank == 1) ? 'V' : ResultAlgebra);
639 // resulting layout policy
640 using r_layout_p = typename detail::layout_to_policy<std::decay_t<decltype(idxm)>>::type;
641 return basic_array_view<r_v_t, res_rank, r_layout_p, newAlgebra, AccessorPolicy, OwningPolicy>{std::move(idxm), {self.sto, offset}};
642 }
643 }
644}

755// Make an iterator for the view/array depending on its type.
756template <typename Iterator>
757[[nodiscard]] auto make_iterator(bool at_end) const noexcept {
758 if constexpr (iterator_rank == Rank) {
759 // multi-dimensional iterator
760 if constexpr (layout_t::is_stride_order_C()) {
761 // C-order case (array_iterator already traverses the data in C-order)
762 return Iterator{indexmap().lengths(), indexmap().strides(), sto.data(), at_end};
763 } else {
764 // general case (we need to permute the shape and the strides according to the stride order of the layout)
765 return Iterator{nda::permutations::apply(layout_t::stride_order, indexmap().lengths()),
766 nda::permutations::apply(layout_t::stride_order, indexmap().strides()), sto.data(), at_end};
767 }
768 } else {
769 // 1-dimensional iterator
770 return Iterator{std::array<long, 1>{size()}, std::array<long, 1>{indexmap().min_stride()}, sto.data(), at_end};
771 }
772}

882// Implementation of the assignment from an n-dimensional array type.
883template <typename RHS>
884void assign_from_ndarray(RHS const &rhs) { // FIXME noexcept {
885#ifdef NDA_ENFORCE_BOUNDCHECK
886 if (this->shape() != rhs.shape())
887 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Size mismatch:"
888 << "\n LHS.shape() = " << this->shape() << "\n RHS.shape() = " << rhs.shape();
889#endif
890 // compile-time check if assignment is possible
891 static_assert(std::is_assignable_v<value_type &, get_value_t<RHS>>, "Error in assign_from_ndarray: Incompatible value types");
892
893 // are both operands nda::MemoryArray types?
894 static constexpr bool both_in_memory = MemoryArray<self_t> and MemoryArray<RHS>;
895
896 // do both operands have the same stride order?
897 static constexpr bool same_stride_order = get_layout_info<self_t>.stride_order == get_layout_info<RHS>.stride_order;
898
899 // prefer optimized options if possible
900 if constexpr (both_in_memory and same_stride_order) {
901 if (rhs.empty()) return;
902 // are both operands strided in 1d?
903 static constexpr bool both_1d_strided = has_layout_strided_1d<self_t> and has_layout_strided_1d<RHS>;
904 if constexpr (mem::on_host<self_t, RHS> and both_1d_strided) {
905 // vectorizable copy on host
906//TODO:: assignment of lazy
907 for (long i = 0; i < size(); ++i) (*this)(_linear_index_t{i}) = rhs(_linear_index_t{i});
908 return;
909 } else if constexpr (!mem::on_host<self_t, RHS> and have_same_value_type_v<self_t, RHS>) {
910 // check for block-layout and use mem::memcpy2D if possible
911 auto bl_layout_dst = get_block_layout(*this);
912 auto bl_layout_src = get_block_layout(rhs);
913 if (bl_layout_dst && bl_layout_src) {
914 auto [n_bl_dst, bl_size_dst, bl_str_dst] = *bl_layout_dst;
915 auto [n_bl_src, bl_size_src, bl_str_src] = *bl_layout_src;
916 // check that the total memory size is the same
917 if (n_bl_dst * bl_size_dst != n_bl_src * bl_size_src) NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Incompatible block sizes";
918 // if either destination or source consists of a single block, we can chunk it up to make the layouts compatible
919 if (n_bl_dst == 1 && n_bl_src > 1) {
920 n_bl_dst = n_bl_src;
921 bl_size_dst /= n_bl_src;
922 bl_str_dst = bl_size_dst;
923 }
924 if (n_bl_src == 1 && n_bl_dst > 1) {
925 n_bl_src = n_bl_dst;
926 bl_size_src /= n_bl_dst;
927 bl_str_src = bl_size_src;
928 }
929 // copy only if block-layouts are compatible, otherwise continue to fallback
930 if (n_bl_dst == n_bl_src && bl_size_dst == bl_size_src) {
931 mem::memcpy2D<mem::get_addr_space<self_t>, mem::get_addr_space<RHS>>((void *)data(), bl_str_dst * sizeof(value_type), (void *)rhs.data(),
932 bl_str_src * sizeof(value_type), bl_size_src * sizeof(value_type),
933 n_bl_src);
934 return;
935 }
936 }
937 }
938 }
939 // otherwise fallback to elementwise assignment
940 if constexpr (mem::on_device<self_t> || mem::on_device<RHS>) {
941 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Fallback to elementwise assignment not implemented for arrays/views on the GPU";
942 }
943 if constexpr (is_aligned and Vectorizable<ValueType>) {

956// Implementation to fill a view/array with a constant scalar value.
957template <typename Scalar>
958void fill_with_scalar(Scalar const &scalar) noexcept {
959 // we make a special implementation if the array is strided in 1d or contiguous
960 if constexpr (has_layout_strided_1d<self_t>) {
961 const long L = size();
962 auto *__restrict const p = data(); // no alias possible here!
963 if constexpr (has_contiguous_layout<self_t>) {
964 for (long i = 0; i < L; ++i) p[i] = scalar;
965 } else {
966 const long stri = indexmap().min_stride();
967 const long Lstri = L * stri;
968 for (long i = 0; i != Lstri; i += stri) p[i] = scalar;
969 }
970 } else {
971 // no compile-time memory layout guarantees
972 for (auto &x : *this) x = scalar;
973 }
974}

978void assign_from_scalar(Scalar const &scalar) noexcept {
979 static_assert(!is_const, "Error in assign_from_ndarray: Cannot assign to a const view");
980 if constexpr (Algebra != 'M') {
981 // element-wise assignment for non-matrix algebras
982 fill_with_scalar(scalar);
983 } else {
984 // a scalar has to be interpreted as a unit matrix for matrix algebras (the scalar in the shortest diagonal)
985 // FIXME : A priori faster to put 0 everywhere and then change the diag to avoid the if.
986 // FIXME : Benchmark and confirm.
987 if constexpr (is_scalar_or_convertible_v<Scalar>)
988 fill_with_scalar(0);
989 else
990 fill_with_scalar(Scalar{0 * scalar}); // FIXME : improve this
991 const long imax = std::min(extent(0), extent(1));
992 for (long i = 0; i < imax; ++i) operator()(i, i) = scalar;
993 }

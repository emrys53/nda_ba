607template <char ResultAlgebra, bool SelfIsRvalue, typename Self, typename... Ts>
608FORCEINLINE static decltype(auto) call(Self &&self, Ts const &...idxs) noexcept(has_no_boundcheck) {
609 // resulting value type
610 using r_v_t = std::conditional_t<std::is_const_v<std::remove_reference_t<Self>>, ValueType const, ValueType>;
611
612 // behavior depends on the given arguments
613 if constexpr (clef::is_any_lazy<Ts...>) {
614 // if there are lazy arguments, e.g. as in A(i_) << i_, a lazy expression is returned
615 return clef::make_expr_call(std::forward<Self>(self), idxs...);
616 } else if constexpr (sizeof...(Ts) == 0) {
617 // if no arguments are given, a full view is returned
618 return basic_array_view<r_v_t, Rank, LayoutPolicy, Algebra, AccessorPolicy, OwningPolicy>{self.lay, self.sto};
619 } else {
620 // otherwise we check the arguments and either access a single element or make a slice
621 static_assert(((layout_t::template argument_is_allowed_for_call_or_slice<Ts> + ...) > 0),
622 "Error in array/view: Slice arguments must be convertible to range, ellipsis, or long (or string if the layout permits it)");
623
624 // number of arguments convertible to long
625 static constexpr int n_args_long = (layout_t::template argument_is_allowed_for_call<Ts> + ...);
626
627 if constexpr (n_args_long == rank) {
628 // access a single element
629 long offset = self.lay(idxs...);
630 if constexpr (is_view or not SelfIsRvalue) {
631 // if the calling object is a view or an lvalue, we return a reference
632 return AccessorPolicy::template accessor<r_v_t>::access(self.sto.data(), offset);
633 } else {
634 // otherwise, we return a copy of the value
635 return ValueType{self.sto[offset]};
636 }
637 } else {
638 // access a slice of the view/array
639 auto const [offset, idxm] = self.lay.slice(idxs...);
640 static constexpr auto res_rank = decltype(idxm)::rank();
641 // resulting algebra
642 static constexpr char newAlgebra = (ResultAlgebra == 'M' and (res_rank == 1) ? 'V' : ResultAlgebra);
643 // resulting layout policy
644 using r_layout_p = typename detail::layout_to_policy<std::decay_t<decltype(idxm)>>::type;
645 return basic_array_view<r_v_t, res_rank, r_layout_p, newAlgebra, AccessorPolicy, OwningPolicy>{std::move(idxm), {self.sto, offset}};
646 }
647 }
648}

699FORCEINLINE native_simd<ValueType> load(Args... idx) const {
700 static_assert(Vectorizable<ValueType>, "Load function is called with a type that is not a vectorizable type");
701 const long offset = lay(idx...);
702 if constexpr(is_aligned and is_padded) {
703 return native_simd<ValueType>(data()+offset);

771// Make an iterator for the view/array depending on its type.
772template <typename Iterator>
773[[nodiscard]] auto make_iterator(bool at_end) const noexcept {
774 if constexpr (iterator_rank == Rank) {
775 // multi-dimensional iterator
776 if constexpr (layout_t::is_stride_order_C()) {
777 // C-order case (array_iterator already traverses the data in C-order)
778 return Iterator{indexmap().lengths(), indexmap().strides(), sto.data(), at_end};
779 } else {
780 // general case (we need to permute the shape and the strides according to the stride order of the layout)
781 return Iterator{nda::permutations::apply(layout_t::stride_order, indexmap().lengths()),
782 nda::permutations::apply(layout_t::stride_order, indexmap().strides()), sto.data(), at_end};
783 }
784 } else {
785 // 1-dimensional iterator
786 return Iterator{std::array<long, 1>{size()}, std::array<long, 1>{indexmap().min_stride()}, sto.data(), at_end};
787 }
788}

898// Implementation of the assignment from an n-dimensional array type.
899template <typename RHS>
900void assign_from_ndarray(RHS const &rhs) { // FIXME noexcept {
901#ifdef NDA_ENFORCE_BOUNDCHECK
902 if (this->shape() != rhs.shape())
903 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Size mismatch:"
904 << "\n LHS.shape() = " << this->shape() << "\n RHS.shape() = " << rhs.shape();
905#endif
906 // compile-time check if assignment is possible
907 static_assert(std::is_assignable_v<value_type &, get_value_t<RHS>>, "Error in assign_from_ndarray: Incompatible value types");
908
909 // are both operands nda::MemoryArray types?
910 static constexpr bool both_in_memory = MemoryArray<self_t> and MemoryArray<RHS>;
911
912 // do both operands have the same stride order?
913 static constexpr bool same_stride_order = get_layout_info<self_t>.stride_order == get_layout_info<RHS>.stride_order;
914
915 // prefer optimized options if possible
916 if constexpr (both_in_memory and same_stride_order) {
917 if (rhs.empty()) return;
918 // are both operands strided in 1d?
919 static constexpr bool both_1d_strided = has_layout_strided_1d<self_t> and has_layout_strided_1d<RHS>;
920 if constexpr (mem::on_host<self_t, RHS> and both_1d_strided) {
921 // vectorizable copy on host
922//TODO:: assignment of lazy
923 for (long i = 0; i < size(); ++i) (*this)(_linear_index_t{i}) = rhs(_linear_index_t{i});
924 return;
925 } else if constexpr (!mem::on_host<self_t, RHS> and have_same_value_type_v<self_t, RHS>) {
926 // check for block-layout and use mem::memcpy2D if possible
927 auto bl_layout_dst = get_block_layout(*this);
928 auto bl_layout_src = get_block_layout(rhs);
929 if (bl_layout_dst && bl_layout_src) {
930 auto [n_bl_dst, bl_size_dst, bl_str_dst] = *bl_layout_dst;
931 auto [n_bl_src, bl_size_src, bl_str_src] = *bl_layout_src;
932 // check that the total memory size is the same
933 if (n_bl_dst * bl_size_dst != n_bl_src * bl_size_src) NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Incompatible block sizes";
934 // if either destination or source consists of a single block, we can chunk it up to make the layouts compatible
935 if (n_bl_dst == 1 && n_bl_src > 1) {
936 n_bl_dst = n_bl_src;
937 bl_size_dst /= n_bl_src;
938 bl_str_dst = bl_size_dst;
939 }
940 if (n_bl_src == 1 && n_bl_dst > 1) {
941 n_bl_src = n_bl_dst;
942 bl_size_src /= n_bl_dst;
943 bl_str_src = bl_size_src;
944 }
945 // copy only if block-layouts are compatible, otherwise continue to fallback
946 if (n_bl_dst == n_bl_src && bl_size_dst == bl_size_src) {
947 mem::memcpy2D<mem::get_addr_space<self_t>, mem::get_addr_space<RHS>>((void *)data(), bl_str_dst * sizeof(value_type), (void *)rhs.data(),
948 bl_str_src * sizeof(value_type), bl_size_src * sizeof(value_type),
949 n_bl_src);
950 return;
951 }
952 }
953 }
954 }
955 // otherwise fallback to elementwise assignment
956 if constexpr (mem::on_device<self_t> || mem::on_device<RHS>) {
957 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Fallback to elementwise assignment not implemented for arrays/views on the GPU";
958 }
959 //TODO:check if their layout are compatible. And call the for_each with specific layout.
960 if constexpr (same_stride_order and Vectorizable<ValueType> and is_simd_enabled_v2_v<ValueType, RHS> and (get_layout_info<self_t>.stride_order == C_stride_order<Rank>) and (has_contiguous_layout<self_t> and has_contiguous_layout<RHS>)) {
961 nda::for_each(shape(),[this, &rhs](auto const &...args) {(*this).store(rhs.load(args...), args...); }, [this, &rhs](auto const &...args) { (*this)(args...) = rhs(args...); }, native_simd<ValueType>::size());
962 }
963 else {
964 nda::for_each(shape(), [this, &rhs](auto const &...args) { (*this)(args...) = rhs(args...); });
965 }
966}

968// Implementation to fill a view/array with a constant scalar value.
969template <typename Scalar>
970void fill_with_scalar(Scalar const &scalar) noexcept {
971 // we make a special implementation if the array is strided in 1d or contiguous
972 if constexpr (has_layout_strided_1d<self_t>) {
973 const long L = size();
974 auto *__restrict const p = data(); // no alias possible here!
975 if constexpr (has_contiguous_layout<self_t>) {
976 for (long i = 0; i < L; ++i) p[i] = scalar;
977 } else {
978 const long stri = indexmap().min_stride();
979 const long Lstri = L * stri;
980 for (long i = 0; i != Lstri; i += stri) p[i] = scalar;
981 }
982 } else {
983 // no compile-time memory layout guarantees
984 for (auto &x : *this) x = scalar;
985 }
986}

990void assign_from_scalar(Scalar const &scalar) noexcept {
991 static_assert(!is_const, "Error in assign_from_ndarray: Cannot assign to a const view");
992 if constexpr (Algebra != 'M') {
993 // element-wise assignment for non-matrix algebras
994 fill_with_scalar(scalar);
995 } else {
996 // a scalar has to be interpreted as a unit matrix for matrix algebras (the scalar in the shortest diagonal)
997 // FIXME : A priori faster to put 0 everywhere and then change the diag to avoid the if.
998 // FIXME : Benchmark and confirm.
999 if constexpr (is_scalar_or_convertible_v<Scalar>)
1000 fill_with_scalar(0);
1001 else
1002 fill_with_scalar(Scalar{0 * scalar}); // FIXME : improve this
1003 const long imax = std::min(extent(0), extent(1));
1004 for (long i = 0; i < imax; ++i) operator()(i, i) = scalar;
1005 }

169 [[nodiscard]] long capacity() const noexcept {
170 size_t required_padding = padding.get_padding();
171 if (required_padding == 0) { return size(); }
172 long cap = mem::next_multiple(len[stride_order[Rank - 1]], required_padding);
173 for (int i = Rank - 2; i >= 0; --i) { cap *= len[stride_order[i]]; }

273 template <std::integral Int>
274 [[nodiscard]] static bool is_stride_order_valid(Int *lenptr, Int *strptr) {
275 auto dims_to_check = std::vector<int>{};
276 dims_to_check.reserve(Rank);
277 for (auto dim : stride_order)
278 if (lenptr[dim] > 1) dims_to_check.push_back(dim);
279 for (int n = 1; n < dims_to_check.size(); ++n)
280 if (std::abs(strptr[dims_to_check[n - 1]]) < std::abs(strptr[dims_to_check[n]])) return false;
281 return true;
282 }

292 // Compute contiguous strides from the shape.
293 void compute_strides_contiguous() {
294 long s = 1;
295 for (int v = rank() - 1; v >= 0; --v) {
296 int u = stride_order[v];
297 str[u] = s;
298 s *= len[u];
299 }
300 ENSURES(s == size());
301 }

329 void init_default_str() {
330 // If padding is not 0, that means we need padding. Pad the fastest dimension array and default_stride should be padded_str to access memory.
331 // Else there is no padding default_str is just the original str.
332 if (get_padding() != 0) {
333 // Padding exists only in aligned memory layout therefor minimum stride has to be 1.
334 padded_str[stride_order[Rank - 1]] = 1;
335 long init = mem::next_multiple(len[stride_order[Rank - 1]], padding.get_padding());
336 for (int i = rank() - 2; i >= 0; --i) {
337 padded_str[stride_order[i]] = init;
338 init *= len[stride_order[i]];
339 }
340 default_str = padded_str;
341 } else {
342 default_str = str;
343 }
344 }

557 template <bool skip_stride, auto I>
558 [[nodiscard]] FORCEINLINE long myget(ellipsis) const noexcept {
559 // nda::ellipsis are skipped and do not contribute to the linear index
560 return 0;

563 // Get the contribution to the linear index in case of a long argument.
564 template <bool skip_stride, auto I>
565 [[nodiscard]] FORCEINLINE long myget(long arg) const noexcept {
566 if constexpr (skip_stride and (I == stride_order[Rank - 1])) {
567 // optimize for the case when the fastest varying dimension is contiguous in memory
568 return arg;
569 } else {
570 // otherwise multiply the argument by the stride of the current dimension
571 return arg * std::get<I>(strides());
572 }
573 }
574
575 // Is the smallest stride equal to one, i.e. is the fastest varying dimension contiguous in memory?
576 static constexpr bool smallest_stride_is_one = has_smallest_stride_is_one(layout_prop);
577
578 // Implementation of the function call operator that takes a multi-dimensional index and maps it to a linear index.
579 template <typename... Args, size_t... Is>
580 [[nodiscard]] FORCEINLINE long call_impl(std::index_sequence<Is...>, Args... args) const noexcept {
581 // set e_pos to the position of the ellipsis, otherwise to -1
582 static constexpr int e_pos = ((std::is_same_v<Args, ellipsis> ? int(Is) + 1 : 0) + ...) - 1;
583
584 if constexpr (e_pos == -1) {
585 // no ellipsis present
586 if constexpr (smallest_stride_is_one) {
587 // optimize for the case that the fastest varying dimension is contiguous in memory
588 return (myget<true, Is>(static_cast<long>(args)) + ...);
589 } else {
590 // arbitrary layouts
591 return ((args * std::get<Is>(default_str)) + ...);
592 }
593 } else {
594 // empty ellipsis is present and needs to be skipped
595 return (myget<smallest_stride_is_one, (Is < e_pos ? Is : Is - 1)>(args) + ...);
596 }
597 }

650 std::array<long, Rank> to_idx(long lin_idx) const {
651 // compute residues starting from slowest index
652 std::array<long, Rank> residues;
653 residues[0] = lin_idx;
654 for (auto i : range(1, Rank)) { residues[i] = residues[i - 1] % default_str[stride_order[i - 1]]; }
655
656 // convert residues to indices, ordered from slowest to fastest
657 std::array<long, Rank> idx;
658 idx[Rank - 1] = residues[Rank - 1] / default_str[stride_order[Rank - 1]];
659 for (auto i : range(Rank - 2, -1, -1)) { idx[i] = (residues[i] - residues[i + 1]) / default_str[stride_order[i]]; }
660
661 // reorder indices according to stride order
662 return permutations::apply_inverse(stride_order, idx);
663 }

710 template <uint64_t Permutation>
711 auto transpose() const {
712 // Makes a new transposed idx_map with permutation P such that
713 // denoting here A = this, A' = P A = returned_value
714 // A'(i_k) = A(i_{P[k]})
715 //
716 // Note that this convention is the correct one to have a (left) action of the symmetric group on
717 // a array and it may not be completely obvious.
718 // Proof
719 // let's operate with P then Q, and denote A'' = Q A'. We want to show that A'' = (QP) A
720 // A'(i_k) = A(i_{P[k]})
721 // A''(j_k) = A'(j_{Q[k]})
722 // then i_k = j_{Q[k]} and A''(j_k) = A(i_{P[k]}) = A(j_{Q[P[k]]}) = A(j_{(QP)[k]}), q.e.d
723 //
724 // NB test will test this composition
725 // Denoting this as A, an indexmap, calling it returns the linear index given by
726 //
727 // A(i_k) = sum_k i_k * S[k] (1)
728 //
729 // where S[k] denotes the strides.
730 //
731 // 1- S' : strides of A'
732 // A'(i_k) = sum_k i_{P[k]} * S[k] = sum_k i_k * S[P{^-1}[k]]
733 // so
734 // S'[k] = S[P{^-1}[k]] (2)
735 // i.e. apply (inverse(P), S) or apply_inverse directly.
736 //
737 // 2- L' : lengths of A'
738 // if L[k] is the k-th length, then because of the definition of A', i.e. A'(i_k) = A(i_{P[k]})
739 // i_q in the lhs A is at position q' such that P[q'] = q (A'(i0 i1 i2...) = A( i_P0 i_P1 i_P2....)
740 // hence L'[q] = L[q'] = L[P^{-1}[q]]
741 // same for static length
742 //
743 // 3- stride_order: denoted in this paragraph as Q (and Q' for A').
744 // by definition Q is a permutation such that Q[0] is the slowest index, Q[Rank -1] the fastest
745 // hence S[Q[k]] is a strictly decreasing sequence (as checked by strides_compatible_to_stride_order)
746 // we want therefore Q' the permutation that will sort the S', i.e.
747 // S'[Q'[k]] = S[Q[k]]
748 // using (2), we have S[P{^-1}[Q'[k]]] = S[Q[k]]
749 // so the permutation Q' is such that P{^-1}Q' = Q or Q' = PQ (as permutation product/composition).
750 // NB : Q and P are permutations, so the operation must be a composition, not an apply (apply applies
751 // a P to any set, like L, S, not only a permutation) even though they are all std::array in the code ...
752 //
753 static constexpr std::array<int, Rank> permu = decode<Rank>(Permutation);
754 static constexpr std::array<int, Rank> new_stride_order = permutations::compose(permu, stride_order);
755 static constexpr std::array<int, Rank> new_static_extents = permutations::apply_inverse(permu, static_extents);
756 //TODO: Think more later whether to use default_str or str.
757 return idx_map<Rank, encode(new_static_extents), encode(new_stride_order), LayoutProp>{permutations::apply_inverse(permu, lengths()),
758 permutations::apply_inverse(permu, strides())};
759 }

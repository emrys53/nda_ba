157 [[nodiscard]] long capacity() const noexcept {
158 size_t required_padding = padding.get_padding();
159 if (required_padding == 0) { return size(); }
160 long cap = mem::next_multiple(len[stride_order[Rank - 1]], required_padding);

261 template <std::integral Int>
262 [[nodiscard]] static bool is_stride_order_valid(Int *lenptr, Int *strptr) {
263 auto dims_to_check = std::vector<int>{};
264 dims_to_check.reserve(Rank);
265 for (auto dim : stride_order)
266 if (lenptr[dim] > 1) dims_to_check.push_back(dim);
267 for (int n = 1; n < dims_to_check.size(); ++n)
268 if (std::abs(strptr[dims_to_check[n - 1]]) < std::abs(strptr[dims_to_check[n]])) return false;
269 return true;
270 }

277 [[nodiscard]] bool is_stride_order_valid() const { return is_stride_order_valid(len.data(), str.data()); }

280 // Compute contiguous strides from the shape.
281 void compute_strides_contiguous() {
282 long s = 1;
283 for (int v = rank() - 1; v >= 0; --v) {
284 int u = stride_order[v];
285 str[u] = s;
286 s *= len[u];
287 }
288 ENSURES(s == size());
289 }

317 void init_default_str() {
318 // If padding is not 0, that means we need padding. Pad the fastest dimension array and default_stride should be padded_str to access memory.
319 // Else there is no padding default_str is just the original str.
320 if (get_padding() != 0) {
321 // Padding exists only in aligned memory layout therefor minimum stride has to be 1.
322 padded_str[stride_order[Rank - 1]] = 1;
323 long init = mem::next_multiple(len[stride_order[Rank - 1]], padding.get_padding());
324 for (int i = rank() - 2; i >= 0; --i) {
325 padded_str[stride_order[i]] = init;
326 init *= len[stride_order[i]];
327 }
328 default_str = padded_str;
329 } else {
330 default_str = str;
331 }
332 }

544 // Get the contribution to the linear index in case of an nda::ellipsis argument.
545 template <bool skip_stride, auto I>
546 [[nodiscard]] FORCEINLINE long myget(ellipsis) const noexcept {
547 // nda::ellipsis are skipped and do not contribute to the linear index
548 return 0;
549 }
550
551 // Get the contribution to the linear index in case of a long argument.
552 template <bool skip_stride, auto I>
553 [[nodiscard]] FORCEINLINE long myget(long arg) const noexcept {
554 if constexpr (skip_stride and (I == stride_order[Rank - 1])) {
555 // optimize for the case when the fastest varying dimension is contiguous in memory
556 return arg;
557 } else {
558 // otherwise multiply the argument by the stride of the current dimension
559 return arg * std::get<I>(strides());
560 }
561 }
562
563 // Is the smallest stride equal to one, i.e. is the fastest varying dimension contiguous in memory?
564 static constexpr bool smallest_stride_is_one = has_smallest_stride_is_one(layout_prop);
565
566 // Implementation of the function call operator that takes a multi-dimensional index and maps it to a linear index.
567 template <typename... Args, size_t... Is>
568 [[nodiscard]] FORCEINLINE long call_impl(std::index_sequence<Is...>, Args... args) const noexcept {
569 // set e_pos to the position of the ellipsis, otherwise to -1
570 static constexpr int e_pos = ((std::is_same_v<Args, ellipsis> ? int(Is) + 1 : 0) + ...) - 1;
571
572 if constexpr (e_pos == -1) {
573 // no ellipsis present
574 if constexpr (smallest_stride_is_one) {
575 // optimize for the case that the fastest varying dimension is contiguous in memory
576 return (myget<true, Is>(static_cast<long>(args)) + ...);
577 } else {
578 // arbitrary layouts
579 return ((args * std::get<Is>(default_str)) + ...);
580 }
581 } else {
582 // empty ellipsis is present and needs to be skipped
583 return (myget<smallest_stride_is_one, (Is < e_pos ? Is : Is - 1)>(args) + ...);
584 }
585 }

638 //TODO: fix this to work with padded/unpadded. Maybe add a struct for it with info idx and strides.
639 std::array<long, Rank> to_idx(long lin_idx) const {
640 // compute residues starting from slowest index
641 std::array<long, Rank> residues;
642 residues[0] = lin_idx;
643 for (auto i : range(1, Rank)) { residues[i] = residues[i - 1] % default_str[stride_order[i - 1]]; }
644
645 // convert residues to indices, ordered from slowest to fastest
646 std::array<long, Rank> idx;
647 idx[Rank - 1] = residues[Rank - 1] / default_str[stride_order[Rank - 1]];
648 for (auto i : range(Rank - 2, -1, -1)) { idx[i] = (residues[i] - residues[i + 1]) / default_str[stride_order[i]]; }
649
650 // reorder indices according to stride order
651 return permutations::apply_inverse(stride_order, idx);
652 }

699 template <uint64_t Permutation>
700 auto transpose() const {
701 // Makes a new transposed idx_map with permutation P such that
702 // denoting here A = this, A' = P A = returned_value
703 // A'(i_k) = A(i_{P[k]})
704 //
705 // Note that this convention is the correct one to have a (left) action of the symmetric group on
706 // a array and it may not be completely obvious.
707 // Proof
708 // let's operate with P then Q, and denote A'' = Q A'. We want to show that A'' = (QP) A
709 // A'(i_k) = A(i_{P[k]})
710 // A''(j_k) = A'(j_{Q[k]})
711 // then i_k = j_{Q[k]} and A''(j_k) = A(i_{P[k]}) = A(j_{Q[P[k]]}) = A(j_{(QP)[k]}), q.e.d
712 //
713 // NB test will test this composition
714 // Denoting this as A, an indexmap, calling it returns the linear index given by
715 //
716 // A(i_k) = sum_k i_k * S[k] (1)
717 //
718 // where S[k] denotes the strides.
719 //
720 // 1- S' : strides of A'
721 // A'(i_k) = sum_k i_{P[k]} * S[k] = sum_k i_k * S[P{^-1}[k]]
722 // so
723 // S'[k] = S[P{^-1}[k]] (2)
724 // i.e. apply (inverse(P), S) or apply_inverse directly.
725 //
726 // 2- L' : lengths of A'
727 // if L[k] is the k-th length, then because of the definition of A', i.e. A'(i_k) = A(i_{P[k]})
728 // i_q in the lhs A is at position q' such that P[q'] = q (A'(i0 i1 i2...) = A( i_P0 i_P1 i_P2....)
729 // hence L'[q] = L[q'] = L[P^{-1}[q]]
730 // same for static length
731 //
732 // 3- stride_order: denoted in this paragraph as Q (and Q' for A').
733 // by definition Q is a permutation such that Q[0] is the slowest index, Q[Rank -1] the fastest
734 // hence S[Q[k]] is a strictly decreasing sequence (as checked by strides_compatible_to_stride_order)
735 // we want therefore Q' the permutation that will sort the S', i.e.
736 // S'[Q'[k]] = S[Q[k]]
737 // using (2), we have S[P{^-1}[Q'[k]]] = S[Q[k]]
738 // so the permutation Q' is such that P{^-1}Q' = Q or Q' = PQ (as permutation product/composition).
739 // NB : Q and P are permutations, so the operation must be a composition, not an apply (apply applies
740 // a P to any set, like L, S, not only a permutation) even though they are all std::array in the code ...
741 //
742 static constexpr std::array<int, Rank> permu = decode<Rank>(Permutation);
743 static constexpr std::array<int, Rank> new_stride_order = permutations::compose(permu, stride_order);
744 static constexpr std::array<int, Rank> new_static_extents = permutations::apply_inverse(permu, static_extents);
745 return idx_map<Rank, encode(new_static_extents), encode(new_stride_order), LayoutProp>{permutations::apply_inverse(permu, lengths()),
746 permutations::apply_inverse(permu, strides()), mem::stride_padding(get_padding())};
747 }

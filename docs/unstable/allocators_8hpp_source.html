186 template <int ChunkSize>
187 class bucket {
188 // Unique pointer to handle memory allocation for the bucket.
189 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
190
191 // Pointer to the start of the bucket.
192 char *p = _start.get();
193
194 // Bitmask to keep track of which chunks are free.
195 uint64_t flags = uint64_t(-1);
196
197 public:
199 static constexpr int TotalChunkSize = 64 * ChunkSize;
200
202 static constexpr auto address_space = Host;
203
204#ifdef NDA_USE_ASAN
205 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
206 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
207#else
209 bucket() = default;
210#endif
212 bucket(bucket const &) = delete;
213
215 bucket(bucket &&) = default;
216
218 bucket &operator=(bucket const &) = delete;
219
221 bucket &operator=(bucket &&) = default;
222
229 blk_t allocate(size_t s) noexcept {
230 // check the size and if there is a free chunk, otherwise abort
231 if (s > ChunkSize) std::abort();
232 if (flags == 0) std::abort();
233
234 // find the first free chunk
235 int pos = __builtin_ctzll(flags);
236
237 // update the bitmask and return the memory block
238 flags &= ~(1ull << pos);
239 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
240#ifdef NDA_USE_ASAN
241 __asan_unpoison_memory_region(b.ptr, ChunkSize);
242#endif
243 return b;
244 }
245
252 blk_t allocate_zero(size_t s) noexcept {
253 auto blk = allocate(s);
254 std::memset(blk.ptr, 0, s);
255 return blk;
256 }
257
262 void deallocate(blk_t b) noexcept {
263#ifdef NDA_USE_ASAN
264 __asan_poison_memory_region(b.ptr, ChunkSize);
265#endif
266 int pos = (b.ptr - p) / ChunkSize;
267 flags |= (1ull << pos);
268 }
269
274 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
275
280 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
281
286 [[nodiscard]] const char *data() const noexcept { return p; }
287
292 [[nodiscard]] auto mask() const noexcept { return flags; }
293
300 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
301 };

313 template <int ChunkSize>
314 class multi_bucket {
315 // Alias for the bucket allocator type.
316 using b_t = bucket<ChunkSize>;
317
318 // Vector of nda::mem::bucket allocators (ordered in memory).
319 std::vector<b_t> bu_vec;
320
321 // Iterator to the current bucket in use.
322 typename std::vector<b_t>::iterator bu;
323
324 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
325 [[gnu::noinline]] void find_non_full_bucket() {
326 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
327 if (bu != bu_vec.end()) return;
328 b_t b;
329 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
330 bu = bu_vec.insert(pos, std::move(b));
331 }
332
333 public:
335 static constexpr auto address_space = Host;
336
338 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
339
341 multi_bucket(multi_bucket const &) = delete;
342
344 multi_bucket(multi_bucket &&) = default;
345
347 multi_bucket &operator=(multi_bucket const &) = delete;
348
350 multi_bucket &operator=(multi_bucket &&) = default;
351
358 blk_t allocate(size_t s) noexcept {
359 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
360 return bu->allocate(s);
361 }
362
370 blk_t allocate_zero(size_t s) noexcept {
371 auto blk = allocate(s);
372 std::memset(blk.ptr, 0, s);
373 return blk;
374 }
375
384 void deallocate(blk_t b) noexcept {
385 // try the current bucket first
386 if (bu != bu_vec.end() and bu->owns(b)) {
387 bu->deallocate(b);
388 return;
389 }
390
391 // otherwise, find the owning bucket in the vector and deallocate
392 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
393 --bu;
394 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
395 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
396 bu->deallocate(b);
397
398 // remove bucket the current bucket if it is empty and not the only one
399 if (!bu->empty()) return;
400 if (bu_vec.size() <= 1) return;
401 bu_vec.erase(bu);
402 bu = bu_vec.end();
403 }
404
409 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
410
415 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
416
423 [[nodiscard]] bool owns(blk_t b) const noexcept {
424 bool res = false;
425 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
426 return res;
427 }
428 };

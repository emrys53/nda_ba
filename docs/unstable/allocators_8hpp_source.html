193 template <int ChunkSize>
194 class bucket {
195 // Unique pointer to handle memory allocation for the bucket.
196 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
197
198 // Pointer to the start of the bucket.
199 char *p = _start.get();
200
201 // Bitmask to keep track of which chunks are free.
202 uint64_t flags = uint64_t(-1);
203
204 public:
206 static constexpr int TotalChunkSize = 64 * ChunkSize;
207
209 static constexpr auto address_space = Host;
210
211#ifdef NDA_USE_ASAN
212 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
213 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
214#else
216 bucket() = default;
217#endif
219 bucket(bucket const &) = delete;
220
222 bucket(bucket &&) = default;
223
225 bucket &operator=(bucket const &) = delete;
226
228 bucket &operator=(bucket &&) = default;
229
236 blk_t allocate(size_t s) noexcept {
237 // check the size and if there is a free chunk, otherwise abort
238 if (s > ChunkSize) std::abort();
239 if (flags == 0) std::abort();
240
241 // find the first free chunk
242 int pos = __builtin_ctzll(flags);
243
244 // update the bitmask and return the memory block
245 flags &= ~(1ull << pos);
246 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
247#ifdef NDA_USE_ASAN
248 __asan_unpoison_memory_region(b.ptr, ChunkSize);
249#endif
250 return b;
251 }
252
259 blk_t allocate_zero(size_t s) noexcept {
260 auto blk = allocate(s);
261 std::memset(blk.ptr, 0, s);
262 return blk;
263 }
264
269 void deallocate(blk_t b) noexcept {
270#ifdef NDA_USE_ASAN
271 __asan_poison_memory_region(b.ptr, ChunkSize);
272#endif
273 int pos = (b.ptr - p) / ChunkSize;
274 flags |= (1ull << pos);
275 }
276
281 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
282
287 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
288
293 [[nodiscard]] const char *data() const noexcept { return p; }
294
299 [[nodiscard]] auto mask() const noexcept { return flags; }
300
307 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
308 };

320 template <int ChunkSize>
321 class multi_bucket {
322 // Alias for the bucket allocator type.
323 using b_t = bucket<ChunkSize>;
324
325 // Vector of nda::mem::bucket allocators (ordered in memory).
326 std::vector<b_t> bu_vec;
327
328 // Iterator to the current bucket in use.
329 typename std::vector<b_t>::iterator bu;
330
331 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
332 [[gnu::noinline]] void find_non_full_bucket() {
333 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
334 if (bu != bu_vec.end()) return;
335 b_t b;
336 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
337 bu = bu_vec.insert(pos, std::move(b));
338 }
339
340 public:
342 static constexpr auto address_space = Host;
343
345 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
346
348 multi_bucket(multi_bucket const &) = delete;
349
351 multi_bucket(multi_bucket &&) = default;
352
354 multi_bucket &operator=(multi_bucket const &) = delete;
355
357 multi_bucket &operator=(multi_bucket &&) = default;
358
365 blk_t allocate(size_t s) noexcept {
366 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
367 return bu->allocate(s);
368 }
369
377 blk_t allocate_zero(size_t s) noexcept {
378 auto blk = allocate(s);
379 std::memset(blk.ptr, 0, s);
380 return blk;
381 }
382
391 void deallocate(blk_t b) noexcept {
392 // try the current bucket first
393 if (bu != bu_vec.end() and bu->owns(b)) {
394 bu->deallocate(b);
395 return;
396 }
397
398 // otherwise, find the owning bucket in the vector and deallocate
399 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
400 --bu;
401 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
402 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
403 bu->deallocate(b);
404
405 // remove bucket the current bucket if it is empty and not the only one
406 if (!bu->empty()) return;
407 if (bu_vec.size() <= 1) return;
408 bu_vec.erase(bu);
409 bu = bu_vec.end();
410 }
411
416 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
417
422 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
423
430 [[nodiscard]] bool owns(blk_t b) const noexcept {
431 bool res = false;
432 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
433 return res;
434 }
435 };

548 ~leak_check() {

552 std::abort();

563 blk_t allocate(size_t s) {
564 blk_t b = A::allocate(s);
565 memory_used += b.s;
566 return b;
567 }
568
575 blk_t allocate_zero(size_t s) {
576 blk_t b = A::allocate_zero(s);
577 memory_used += b.s;
578 return b;
579 }
580
586 void deallocate(blk_t b) noexcept {
587 memory_used -= b.s;
588 if (memory_used < 0) {
589#ifndef NDEBUG
590 std::cerr << "Memory used by allocator < 0: Memory block to be deleted: b.s = " << b.s << ", b.ptr = " << (void *)b.ptr << "\n";
591 std::abort();
592#endif
593 }
594 A::deallocate(b);
595 }
596
601 [[nodiscard]] bool empty() const { return (memory_used == 0); }
602
609 [[nodiscard]] bool owns(blk_t b) const noexcept { return A::owns(b); }
610
615 [[nodiscard]] long get_memory_used() const noexcept { return memory_used; }

191 template <int ChunkSize>
192 class bucket {
193 // Unique pointer to handle memory allocation for the bucket.
194 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
195
196 // Pointer to the start of the bucket.
197 char *p = _start.get();
198
199 // Bitmask to keep track of which chunks are free.
200 uint64_t flags = uint64_t(-1);
201
202 public:
204 static constexpr int TotalChunkSize = 64 * ChunkSize;
205
207 static constexpr auto address_space = Host;
208
209#ifdef NDA_USE_ASAN
210 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
211 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
212#else
214 bucket() = default;
215#endif
217 bucket(bucket const &) = delete;
218
220 bucket(bucket &&) = default;
221
223 bucket &operator=(bucket const &) = delete;
224
226 bucket &operator=(bucket &&) = default;
227
234 blk_t allocate(size_t s) noexcept {
235 // check the size and if there is a free chunk, otherwise abort
236 if (s > ChunkSize) std::abort();
237 if (flags == 0) std::abort();
238
239 // find the first free chunk
240 int pos = __builtin_ctzll(flags);
241
242 // update the bitmask and return the memory block
243 flags &= ~(1ull << pos);
244 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
245#ifdef NDA_USE_ASAN
246 __asan_unpoison_memory_region(b.ptr, ChunkSize);
247#endif
248 return b;
249 }
250
257 blk_t allocate_zero(size_t s) noexcept {
258 auto blk = allocate(s);
259 std::memset(blk.ptr, 0, s);
260 return blk;
261 }
262
267 void deallocate(blk_t b) noexcept {
268#ifdef NDA_USE_ASAN
269 __asan_poison_memory_region(b.ptr, ChunkSize);
270#endif
271 int pos = (b.ptr - p) / ChunkSize;
272 flags |= (1ull << pos);
273 }
274
279 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
280
285 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
286
291 [[nodiscard]] const char *data() const noexcept { return p; }
292
297 [[nodiscard]] auto mask() const noexcept { return flags; }
298
305 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
306 };

318 template <int ChunkSize>
319 class multi_bucket {
320 // Alias for the bucket allocator type.
321 using b_t = bucket<ChunkSize>;
322
323 // Vector of nda::mem::bucket allocators (ordered in memory).
324 std::vector<b_t> bu_vec;
325
326 // Iterator to the current bucket in use.
327 typename std::vector<b_t>::iterator bu;
328
329 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
330 [[gnu::noinline]] void find_non_full_bucket() {
331 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
332 if (bu != bu_vec.end()) return;
333 b_t b;
334 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
335 bu = bu_vec.insert(pos, std::move(b));
336 }
337
338 public:
340 static constexpr auto address_space = Host;
341
343 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
344
346 multi_bucket(multi_bucket const &) = delete;
347
349 multi_bucket(multi_bucket &&) = default;
350
352 multi_bucket &operator=(multi_bucket const &) = delete;
353
355 multi_bucket &operator=(multi_bucket &&) = default;
356
363 blk_t allocate(size_t s) noexcept {
364 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
365 return bu->allocate(s);
366 }
367
375 blk_t allocate_zero(size_t s) noexcept {
376 auto blk = allocate(s);
377 std::memset(blk.ptr, 0, s);
378 return blk;
379 }
380
389 void deallocate(blk_t b) noexcept {
390 // try the current bucket first
391 if (bu != bu_vec.end() and bu->owns(b)) {
392 bu->deallocate(b);
393 return;
394 }
395
396 // otherwise, find the owning bucket in the vector and deallocate
397 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
398 --bu;
399 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
400 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
401 bu->deallocate(b);
402
403 // remove bucket the current bucket if it is empty and not the only one
404 if (!bu->empty()) return;
405 if (bu_vec.size() <= 1) return;
406 bu_vec.erase(bu);
407 bu = bu_vec.end();
408 }
409
414 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
415
420 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
421
428 [[nodiscard]] bool owns(blk_t b) const noexcept {
429 bool res = false;
430 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
431 return res;
432 }
433 };

585 memory_used -= b.s;

587#ifndef NDEBUG
588 std::cerr << "Memory used by allocator < 0: Memory block to be deleted: b.s = " << b.s << ", b.ptr = " << (void *)b.ptr << "\n";
589 std::abort();
590#endif
591 }
592 A::deallocate(b);
593 }

600
607 [[nodiscard]] bool owns(blk_t b) const noexcept { return A::owns(b); }
608
613 [[nodiscard]] long get_memory_used() const noexcept { return memory_used; }

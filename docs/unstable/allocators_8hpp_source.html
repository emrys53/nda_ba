203 template <int ChunkSize>
204 class bucket {
205 // Unique pointer to handle memory allocation for the bucket.
206 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
207
208 // Pointer to the start of the bucket.
209 char *p = _start.get();
210
211 // Bitmask to keep track of which chunks are free.
212 uint64_t flags = uint64_t(-1);
213
214 public:
216 static constexpr int TotalChunkSize = 64 * ChunkSize;
217
219 static constexpr auto address_space = Host;
220
221 static constexpr bool is_aligned = false;
222
223#ifdef NDA_USE_ASAN
224 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
225 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
226#else
228 bucket() = default;
229#endif
231 bucket(bucket const &) = delete;
232
234 bucket(bucket &&) = default;
235
237 bucket &operator=(bucket const &) = delete;
238
240 bucket &operator=(bucket &&) = default;
241
248 blk_t allocate(size_t s, [[maybe_unused]] size_t a = 0) noexcept {
249 // check the size and if there is a free chunk, otherwise abort
250 if (s > ChunkSize) std::abort();
251 if (flags == 0) std::abort();
252
253 // find the first free chunk
254 int pos = __builtin_ctzll(flags);
255
256 // update the bitmask and return the memory block
257 flags &= ~(1ull << pos);
258 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
259#ifdef NDA_USE_ASAN
260 __asan_unpoison_memory_region(b.ptr, ChunkSize);
261#endif
262 return b;
263 }
264
271 blk_t allocate_zero(size_t s, size_t a = 0) noexcept {
272 auto blk = allocate(s, a);
273 std::memset(blk.ptr, 0, s);
274 return blk;
275 }
276
281 void deallocate(blk_t b) noexcept {
282#ifdef NDA_USE_ASAN
283 __asan_poison_memory_region(b.ptr, ChunkSize);
284#endif
285 int pos = (b.ptr - p) / ChunkSize;
286 flags |= (1ull << pos);
287 }
288
293 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
294
299 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
300
305 [[nodiscard]] const char *data() const noexcept { return p; }
306
311 [[nodiscard]] auto mask() const noexcept { return flags; }
312
319 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
320 };

332 template <int ChunkSize>
333 class multi_bucket {
334 // Alias for the bucket allocator type.
335 using b_t = bucket<ChunkSize>;
336
337 // Vector of nda::mem::bucket allocators (ordered in memory).
338 std::vector<b_t> bu_vec;
339
340 // Iterator to the current bucket in use.
341 typename std::vector<b_t>::iterator bu;
342
343 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
344 [[gnu::noinline]] void find_non_full_bucket() {
345 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
346 if (bu != bu_vec.end()) return;
347 b_t b;
348 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
349 bu = bu_vec.insert(pos, std::move(b));
350 }
351
352 public:
354 static constexpr auto address_space = Host;
355
356 static constexpr bool is_aligned = false;
357
359 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
360
362 multi_bucket(multi_bucket const &) = delete;
363
365 multi_bucket(multi_bucket &&) = default;
366
368 multi_bucket &operator=(multi_bucket const &) = delete;
369
371 multi_bucket &operator=(multi_bucket &&) = default;
372
379 blk_t allocate(size_t s, size_t a = 0) noexcept {
380 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
381 return bu->allocate(s, a);
382 }
383
391 blk_t allocate_zero(size_t s, size_t a = 0) noexcept {
392 auto blk = allocate(s, a);
393 std::memset(blk.ptr, 0, s);
394 return blk;
395 }
396
405 void deallocate(blk_t b) noexcept {
406 // try the current bucket first
407 if (bu != bu_vec.end() and bu->owns(b)) {
408 bu->deallocate(b);
409 return;
410 }
411
412 // otherwise, find the owning bucket in the vector and deallocate
413 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
414 --bu;
415 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
416 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
417 bu->deallocate(b);
418
419 // remove bucket the current bucket if it is empty and not the only one
420 if (!bu->empty()) return;
421 if (bu_vec.size() <= 1) return;
422 bu_vec.erase(bu);
423 bu = bu_vec.end();
424 }
425
430 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
431
436 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
437
444 [[nodiscard]] bool owns(blk_t b) const noexcept {
445 bool res = false;
446 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
447 return res;
448 }
449 };

566 ~leak_check() {
567 if (!empty()) {
568#ifndef NDEBUG

580 blk_t allocate(size_t s, size_t a = 0) {
581 const size_t prev_size = s;
582 if (a != 0) s = next_multiple(s, a);
583 blk_t b = A::allocate(s, a);
584 memory_used += prev_size;
585 return b;
586 }
587
594 blk_t allocate_zero(size_t s, size_t a = 0) {
595 const size_t prev_size = s;
596 if (a != 0) s = next_multiple(s, a);
597 blk_t b = A::allocate_zero(s, a);
598 memory_used += prev_size;
599 return b;
600 }
601
607 void deallocate(blk_t b) noexcept {
608 memory_used -= b.s;
609 if (memory_used < 0) {
610#ifndef NDEBUG
611 std::cerr << "Memory used by allocator < 0: Memory block to be deleted: b.s = " << b.s << ", b.ptr = " << (void *)b.ptr << "\n";
612 std::abort();
613#endif
614 }
615 A::deallocate(b);
616 }
617
622 [[nodiscard]] bool empty() const { return (memory_used == 0); }
623
630 [[nodiscard]] bool owns(blk_t b) const noexcept { return A::owns(b); }
631
636 [[nodiscard]] long get_memory_used() const noexcept { return memory_used; }
637 };

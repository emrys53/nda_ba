190 template <int ChunkSize>
191 class bucket {
192 // Unique pointer to handle memory allocation for the bucket.
193 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
194
195 // Pointer to the start of the bucket.
196 char *p = _start.get();
197
198 // Bitmask to keep track of which chunks are free.
199 uint64_t flags = uint64_t(-1);
200
201 public:
203 static constexpr int TotalChunkSize = 64 * ChunkSize;
204
206 static constexpr auto address_space = Host;
207
208 static constexpr bool is_aligned = false;
209
210#ifdef NDA_USE_ASAN
211 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
212 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
213#else
215 bucket() = default;
216#endif
218 bucket(bucket const &) = delete;
219
221 bucket(bucket &&) = default;
222
224 bucket &operator=(bucket const &) = delete;
225
227 bucket &operator=(bucket &&) = default;
228
235 blk_t allocate(size_t s, [[maybe_unused]] size_t a = 0) noexcept {
236 // check the size and if there is a free chunk, otherwise abort
237 if (s > ChunkSize) std::abort();
238 if (flags == 0) std::abort();
239
240 // find the first free chunk
241 int pos = __builtin_ctzll(flags);
242
243 // update the bitmask and return the memory block
244 flags &= ~(1ull << pos);
245 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
246#ifdef NDA_USE_ASAN
247 __asan_unpoison_memory_region(b.ptr, ChunkSize);
248#endif
249 return b;
250 }
251
258 blk_t allocate_zero(size_t s, size_t a = 0) noexcept {
259 auto blk = allocate(s, a);
260 std::memset(blk.ptr, 0, s);
261 return blk;
262 }
263
268 void deallocate(blk_t b) noexcept {
269#ifdef NDA_USE_ASAN
270 __asan_poison_memory_region(b.ptr, ChunkSize);
271#endif
272 int pos = (b.ptr - p) / ChunkSize;
273 flags |= (1ull << pos);
274 }
275
280 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
281
286 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
287
292 [[nodiscard]] const char *data() const noexcept { return p; }
293
298 [[nodiscard]] auto mask() const noexcept { return flags; }
299
306 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
307 };

319 template <int ChunkSize>
320 class multi_bucket {
321 // Alias for the bucket allocator type.
322 using b_t = bucket<ChunkSize>;
323
324 // Vector of nda::mem::bucket allocators (ordered in memory).
325 std::vector<b_t> bu_vec;
326
327 // Iterator to the current bucket in use.
328 typename std::vector<b_t>::iterator bu;
329
330 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
331 [[gnu::noinline]] void find_non_full_bucket() {
332 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
333 if (bu != bu_vec.end()) return;
334 b_t b;
335 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
336 bu = bu_vec.insert(pos, std::move(b));
337 }
338
339 public:
341 static constexpr auto address_space = Host;
342
343 static constexpr bool is_aligned = false;
344
346 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
347
349 multi_bucket(multi_bucket const &) = delete;
350
352 multi_bucket(multi_bucket &&) = default;
353
355 multi_bucket &operator=(multi_bucket const &) = delete;
356
358 multi_bucket &operator=(multi_bucket &&) = default;
359
366 blk_t allocate(size_t s, size_t a = 0) noexcept {
367 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
368 return bu->allocate(s, a);
369 }
370
378 blk_t allocate_zero(size_t s, size_t a = 0) noexcept {
379 auto blk = allocate(s, a);
380 std::memset(blk.ptr, 0, s);
381 return blk;
382 }
383
392 void deallocate(blk_t b) noexcept {
393 // try the current bucket first
394 if (bu != bu_vec.end() and bu->owns(b)) {
395 bu->deallocate(b);
396 return;
397 }
398
399 // otherwise, find the owning bucket in the vector and deallocate
400 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
401 --bu;
402 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
403 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
404 bu->deallocate(b);
405
406 // remove bucket the current bucket if it is empty and not the only one
407 if (!bu->empty()) return;
408 if (bu_vec.size() <= 1) return;
409 bu_vec.erase(bu);
410 bu = bu_vec.end();
411 }
412
417 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
418
423 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
424
431 [[nodiscard]] bool owns(blk_t b) const noexcept {
432 bool res = false;
433 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
434 return res;
435 }
436 };

475
477 segregator &operator=(segregator &&) = default;

486 blk_t allocate(size_t s, size_t a = 0) noexcept { return s <= Threshold ? small.allocate(s, a) : big.allocate(s, a); }
487
495 blk_t allocate_zero(size_t s, size_t a = 0) noexcept { return s <= Threshold ? small.allocate_zero(s, a) : big.allocate_zero(s, a); }

553 ~leak_check() {
554 if (!empty()) {
555#ifndef NDEBUG

567 blk_t allocate(size_t s, size_t a = 0) {
568 const size_t prev_size = s;
569 if (a != 0) s = next_multiple(s, a);
570 blk_t b = A::allocate(s, a);
571 memory_used += prev_size;
572 return b;
573 }
574
581 blk_t allocate_zero(size_t s, size_t a = 0) {
582 const size_t prev_size = s;
583 if (a != 0) s = next_multiple(s, a);
584 blk_t b = A::allocate_zero(s, a);
585 memory_used += prev_size;
586 return b;
587 }
588
594 void deallocate(blk_t b) noexcept {
595 memory_used -= b.s;
596 if (memory_used < 0) {
597#ifndef NDEBUG
598 std::cerr << "Memory used by allocator < 0: Memory block to be deleted: b.s = " << b.s << ", b.ptr = " << (void *)b.ptr << "\n";
599 std::abort();
600#endif
601 }
602 A::deallocate(b);
603 }
604
609 [[nodiscard]] bool empty() const { return (memory_used == 0); }
610
617 [[nodiscard]] bool owns(blk_t b) const noexcept { return A::owns(b); }
618
623 [[nodiscard]] long get_memory_used() const noexcept { return memory_used; }

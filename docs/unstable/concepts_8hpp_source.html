192 template <typename H, typename T = typename std::remove_cvref_t<H>::value_type>
193 concept Handle = requires(H const &h) {
194 requires std::is_same_v<typename std::remove_cvref_t<H>::value_type, T>;
195 { h.is_null() } noexcept -> std::same_as<bool>;
196 { h.data() } noexcept -> std::same_as<T *>;
